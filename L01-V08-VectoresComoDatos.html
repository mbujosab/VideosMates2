1.5. Utilidades del vídeo

  • vector(valores): vector columna entre paréntesis (como en el libro) con números enteros o con un decimal; sus componentes se pueden animar una a una (v.get_entries()[k]).
  • leyenda(rotulos, v): la lista de ingredientes (o de países) a la izquierda del vector v, cada rótulo a la altura de su componente.
  • banda(v, k, x0, x1): franja amarilla translúcida sobre la fila k, de x0 a x1: marca que una misma posición significa lo mismo en todos los vectores (y anticipa las filas de una matriz).
  • omega(estado) y bocadillo(...): la criatura y su bocadillo. Cuerpo grueso (boceto A) con la cara dentro del arco (boceto B): ojos, cejas y boca, más un signo según el estado (lámina tmp/bocetos_omega/expresiones.png). Estados: normal (sonriente), neutral (sin sonrisa: para lo que no es motivo de alegría, como un mal dato económico), duda, sorpresa, alegria y meditacion. Gestos: entra, parpadea y cambia (de expresión, sin moverse). Pasará a Comunes.org cuando se fije el diseño.

def vector(valores, color=WHITE, decimales=0, escala=0.8):
    if decimales:
        m = DecimalMatrix([[v] for v in valores], left_bracket="(", right_bracket=")", v_buff=0.62,
                          element_to_mobject_config={"num_decimal_places": decimales, "group_with_commas": False})
    else:
        m = IntegerMatrix([[v] for v in valores], left_bracket="(", right_bracket=")", v_buff=0.62,
                          element_to_mobject_config={"group_with_commas": False})   # 1050, no 1,050
    m.get_entries().set_color(color)
    return m.scale(escala)


def leyenda(rotulos, v, buff=0.35, font_size=30):
    """Rótulos alineados a la derecha, cada uno a la altura de su componente de v."""
    filas = VGroup(*[Tex(r, font_size=font_size) for r in rotulos])
    x_der = v.get_left()[0] - buff
    for f, e in zip(filas, v.get_entries()):
        f.move_to([x_der - f.width/2, e.get_y(), 0])
    return filas


def banda(v, k, x0, x1, color=COLOR_ESCALAR):
    y = v.get_entries()[k].get_y()
    return Rectangle(width=x1 - x0, height=0.5, stroke_width=0, fill_color=color,
                     fill_opacity=0.22).move_to([(x0 + x1)/2, y, 0])


def _cara(estado, alto, mira=ORIGIN):
    """Ojos, cejas y boca de Omega (se colocan dentro del arco). Estados: normal (sonriente),
    neutral (sin sonrisa), duda, sorpresa, alegria y meditacion."""
    u, blanco, r = alto/2, dict(color=WHITE, stroke_width=3), 0.11*alto/2
    def ojo(rr=r, pupila=0.5, m=mira):
        b = Circle(radius=rr, color=WHITE, fill_opacity=1, stroke_width=1.2).set_stroke(BLACK)
        p = Dot(radius=rr*pupila, color=BLACK).move_to(m*rr*0.4)
        return VGroup(b, p, Dot(radius=rr*0.13, color=WHITE).move_to(p.get_center() + rr*0.18*(UP + LEFT)))
    if estado == "alegria":
        ojos = VGroup(*[Arc(radius=0.8*r, start_angle=0, angle=PI, **blanco) for _ in range(2)])
    elif estado == "meditacion":
        ojos = VGroup(*[Arc(radius=0.8*r, start_angle=PI, angle=PI, **blanco) for _ in range(2)])
    elif estado == "sorpresa":
        ojos = VGroup(ojo(1.25*r, 0.3, ORIGIN), ojo(1.25*r, 0.3, ORIGIN))
    elif estado == "duda":
        ojos = VGroup(ojo(m=UR), ojo(m=UR))
    else:
        ojos = VGroup(ojo(), ojo())
    ojos.arrange(RIGHT, buff=0.9*r)
    i, d, w = ojos[0].get_top(), ojos[1].get_top(), 2.2*r
    ceja = lambda c, ang, dy: Line(c + 0.5*w*LEFT, c + 0.5*w*RIGHT, **blanco).rotate(ang).shift((0.12*u + dy)*UP)
    if estado == "duda":          # una ceja alta, con la punta interior levantada
        cejas = VGroup(ceja(i, 0, -0.02*u), ceja(d, -18*DEGREES, 0.1*u))
    elif estado == "sorpresa":
        cejas = VGroup(*[ArcBetweenPoints(c + 0.5*w*LEFT + 0.2*u*UP, c + 0.5*w*RIGHT + 0.2*u*UP, angle=-PI/3, **blanco) for c in (i, d)])
    elif estado in ("meditacion", "neutral"):
        cejas = VGroup(ceja(i, 0, -0.02*u), ceja(d, 0, -0.02*u))
    else:                         # normal y alegría: cejas relajadas
        cejas = VGroup(ceja(i, 10*DEGREES, 0), ceja(d, -10*DEGREES, 0))
    base = ojos.get_bottom() + 0.16*u*DOWN
    if estado == "alegria":
        boca = Arc(radius=0.17*u, start_angle=PI, angle=PI, color=WHITE, fill_color=WHITE, fill_opacity=1, stroke_width=2).move_to(base + 0.04*u*DOWN)
    elif estado == "sorpresa":
        boca = Ellipse(width=0.12*u, height=0.17*u, color=WHITE, stroke_width=3).move_to(base + 0.04*u*DOWN)
    elif estado == "duda":
        boca = Line(base + 0.12*u*LEFT, base + 0.12*u*RIGHT, **blanco).rotate(-12*DEGREES)
    elif estado in ("meditacion", "neutral"):
        boca = Line(base + (0.06 if estado == "meditacion" else 0.1)*u*LEFT, base + (0.06 if estado == "meditacion" else 0.1)*u*RIGHT, **blanco)
    else:
        boca = Arc(radius=0.14*u, start_angle=-PI*0.8, angle=PI*0.6, **blanco).move_to(base)
    return VGroup(ojos, cejas, boca)


def omega(estado="normal", alto=1.6, mira=ORIGIN, color=None):
    """La criatura de la serie: [0] el cuerpo (una Ω gruesa), [1] la cara (ojos, cejas, boca)
    y [2] el signo del estado (?, !, destellos o la nube de pensamiento)."""
    cuerpo = MathTex(r"\boldsymbol{\Omega}").set_color(color or OMEGA_COLOR).scale_to_fit_height(alto)
    cara = _cara(estado, alto, mira).move_to(cuerpo.get_center() + 0.13*alto*UP)
    figura = VGroup(cuerpo, cara)
    signo = VGroup()
    if estado == "duda":
        figura.rotate(-8*DEGREES)
        signo = MathTex("?", font_size=60, color=COLOR_ESCALAR).next_to(figura, UR, buff=0)
    elif estado == "sorpresa":
        figura.stretch(1.08, 1)
        signo = MathTex("!", font_size=60, color=COLOR_ESCALAR).next_to(figura, UR, buff=0)
    elif estado == "alegria":
        figura.stretch(0.95, 1)
        signo = VGroup(*[Star(n=4, outer_radius=0.12, inner_radius=0.04, color=COLOR_ESCALAR, fill_opacity=1).move_to(figura.get_corner(UR) + v)
                         for v in (0.1*UP, 0.35*DOWN + 0.25*RIGHT, 0.25*UP + 0.45*LEFT)])
    elif estado == "meditacion":
        nube = VGroup(*[Circle(radius=rr, color=WHITE, fill_opacity=1) for rr in (0.06, 0.1, 0.38)])
        nube[0].move_to(figura.get_corner(UR) + 0.05*UP); nube[1].next_to(nube[0], UR, buff=0.05); nube[2].next_to(nube[1], UR, buff=0)
        signo = VGroup(nube, MathTex(r"\dots", color=BLACK, font_size=40).move_to(nube[2]))
    o = VGroup(cuerpo, cara, signo)
    o.estado, o.alto, o.mira = estado, alto, mira
    return o


def cambia(escena, criatura, estado, mira=None, run_time=0.6):
    """Cambia la expresión de la criatura sin moverla (el cuerpo se queda donde estaba)."""
    nueva = omega(estado, criatura.alto, criatura.mira if mira is None else mira)
    nueva.shift(criatura[0].get_bottom() - nueva[0].get_bottom())
    # Un signo que aparece o desaparece (?, !, destellos) lo hace en su sitio: transformado
    # desde o hacia un grupo vacío, volaría desde o hacia el origen.
    if len(nueva[2]) == 0 and len(criatura[2]) > 0:
        nueva.submobjects[2] = criatura[2].copy().set_opacity(0)
    elif len(criatura[2]) == 0 and len(nueva[2]) > 0:
        criatura.submobjects[2] = nueva[2].copy().set_opacity(0)
    escena.play(Transform(criatura, nueva), run_time=run_time)
    criatura.estado, criatura.mira = estado, nueva.mira


def bocadillo(texto, criatura, ancho=4.2, font_size=30):
    t = Tex(texto, font_size=font_size, color=BLACK)
    if t.width > ancho:
        t.scale_to_fit_width(ancho)
    globo = RoundedRectangle(corner_radius=0.25, width=t.width + 0.5, height=t.height + 0.4,
                             fill_color=WHITE, fill_opacity=1, stroke_color=WHITE)
    globo.next_to(criatura[0], UL, buff=0.05).shift(0.4*RIGHT)
    t.move_to(globo)
    pico = Polygon(globo.get_corner(DR) + 0.35*LEFT, globo.get_corner(DR) + 0.8*LEFT,
                   criatura[0].get_corner(UL) + 0.15*RIGHT + 0.05*DOWN, fill_color=WHITE, fill_opacity=1, stroke_width=0)
    return VGroup(pico, globo, t)


def parpadea(escena, criatura, run_time=0.3):
    if criatura.estado in ("alegria", "meditacion"):   # ya tiene los ojos cerrados
        return
    ojos = criatura[1][0]
    escena.play(ojos.animate.stretch(0.1, 1), run_time=run_time/2)
    escena.play(ojos.animate.stretch(10, 1), run_time=run_time/2)


def entra(escena, criatura, desde=RIGHT, run_time=1):
    escena.play(FadeIn(criatura, shift=1.5*(-desde)), run_time=run_time)
    parpadea(escena, criatura)

1.6.3. Escena 2 - La cesta de la compra: suma y producto por tres

Arriba, la fórmula de la cesta (se reescribe al multiplicar por tres). Las componentes de las tres recetas vuelan fila a fila hasta la cesta: "patatas con patatas". Antes de multiplicar por tres se quitan las recetas y el igual (si no, la fila diría t+g+e = 3(t+g+e)); un 3 amarillo se pone delante de la cesta y se absorbe en ella. Con la propiedad 5 vuelven las recetas como estaban, se multiplican por tres igual que la cesta (3t, 3g, 3e) y se recupera el igual, resaltado en COLOR_REGLA; la propiedad 5 de V07 aparece en COLOR_REGLA al lado.

3.2. Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la clase del vídeo completo): corregir una escena no obliga a renderizarlo todo ni a pagar otra vez la voz, y los límites entre escenas quedan exactos. Cada segmento se recodifica igualando la duración del audio y del vídeo (concatenar sin recodificar deja un parón en cada unión, pues manim genera el audio unas décimas más largo que la imagen). La música suena en la portada (volumen volumen) y se funde cuando empieza la primera escena; vuelve al empezar la escena cierre (bajo la voz, con volumen_voz), sube en la última escena (los créditos) y se funde a negro con la imagen. La voz se mide entera y se lleva a -16 LUFS con una ganancia fija y un limitador (no con loudnorm en una pasada, que subía el volumen poco a poco al empezar). Los .srt de las escenas se desplazan y se funden en un único video.srt junto al mp4 (para subirlo a YouTube).

escenas es la lista ordenada de clases; cierre la escena en la que vuelve la música. El mp4 y el srt finales quedan en la raíz del repositorio del idioma. 

export LC_ALL=C   # decimales con punto para bc y printf
[ "$idioma" = "en" ] && raiz=../videoMath2 || raiz=.
dir=$raiz/lib/media/videos/$fichero/$calidad
tmp=$raiz/tmp/$video; mkdir -p $tmp; rm -f $tmp/seg_*.mp4

# 1. Un segmento por escena, con audio y vídeo de la misma duración exacta
i=0; lista=""; offsets=""; T=0; T0=""; Tc=""
for esc in $escenas; do
  seg=$(printf "%s/seg_%02d.mp4" $tmp $i)
  dur=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=duration -of default=nk=1:nw=1 $dir/$esc.mp4)
  fps=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=r_frame_rate -of default=nk=1:nw=1 $dir/$esc.mp4)
  if ffprobe -loglevel error -select_streams a -show_entries stream=codec_type -of csv=p=0 $dir/$esc.mp4 | grep -q audio; then
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -af "aresample=48000,apad" -ac 2 -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  else
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -f lavfi -i anullsrc=r=48000:cl=stereo -map 0:v -map 1:a -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  fi
  d=$(ffprobe -loglevel error -show_entries format=duration -of default=nk=1:nw=1 $seg)
  offsets="$offsets $T"; [ "$esc" = "$cierre" ] && Tc=$T
  printf "%2d  %-55s %8.3f s  (empieza en %8.3f)\n" $i $esc $d $T
  T=$(bc -l <<< "$T+$d"); [ -z "$T0" ] && T0=$d; dl=$d
  lista="$lista $seg"; i=$((i+1))
done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
# La voz se mide entera y recibe una ganancia fija hasta -16 LUFS, con un limitador para los picos.
# (loudnorm en una pasada ajusta la ganancia sobre la marcha: la primera frase quedaba 5 dB más baja.)
ia=""; for k in $(seq 0 $((n-1))); do ia="$ia[$k:a]"; done
I=$(ffmpeg -hide_banner -nostats $inputs -filter_complex "${ia}concat=n=$n:v=0:a=1,ebur128=framelog=quiet" -f null - 2>&1 | awk '$1=="I:"{v=$2} END{print v}')
ganancia=$(bc -l <<< "-16 - ($I)"); printf "Voz: %s LUFS; ganancia %.2f dB\n" $I $ganancia
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]volume=${ganancia}dB,alimiter=limit=0.84:level=false[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];

import sys, os, re
dir, salida = sys.argv[1], sys.argv[2]
escenas = os.environ["ESCENAS"].split(); offsets = [float(x) for x in os.environ["OFFSETS"].split()]
def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000
def s2t(x):
    ms = round(x*1000); h, r = divmod(ms, 3600000); m, r = divmod(r, 60000); s, ms = divmod(r, 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"
k, out = 1, []
for esc, off in zip(escenas, offsets):
    ruta = f"{dir}/{esc}.srt"
    if not os.path.exists(ruta): continue
    for bloque in re.split(r"\n\s*\n", open(ruta, encoding="utf-8").read().strip()):
        lineas = bloque.strip().splitlines()
        if len(lineas) < 3 or "-->" not in lineas[1]: continue
        a, _, b = lineas[1].partition("-->")
        out.append(f"{k}\n{s2t(t2s(a)+off)} --> {s2t(t2s(b)+off)}\n" + "\n".join(lineas[2:])); k += 1
open(salida, "w", encoding="utf-8").write("\n\n".join(out) + "\n")
print(f"Subtítulos: {salida} ({k-1} entradas)")
EOF

3.4. Extraer el guion (la narración) para revisarlo

Recorre el fichero de python y saca, escena por escena, los textos tr(...) del idioma pedido: en prosa la narración (los que están dentro de self.voiceover) y entre corchetes los textos que aparecen en pantalla. Sirve para revisar el guion sin leer código y para la descripción o los subtítulos del vídeo. Como los dos py solo difieren en LANG, lee siempre el fichero español (./lib/). 

import ast, textwrap

arbol = ast.parse(open(f"lib/{fichero}.py", encoding="utf-8").read())
idx   = 0 if idioma == "es" else 1

def texto(nodo):
    try:
        return ast.literal_eval(nodo)
    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
