1.5. Utilidades del vídeo

  • vector(valores) y matriz(filas): un vector entre paréntesis y una matriz entre corchetes, con números de un decimal (o enteros con decimales=0); sus componentes se animan una a una (get_entries(), get_rows(), get_columns()).
  • leyenda(rotulos, entradas, x_der): rótulos alineados a la derecha de x_der, cada uno a la altura de su entrada (los países).
  • tabla_pib(centro): la matriz del PIB con los países a la izquierda y los años encima de cada columna.
  • franja_fila y franja_columna: franja amarilla translúcida sobre una fila o una columna (la de las filas, como en V08).
  • con_nombre(tex, mob): la fórmula tex (p. ej. "\Mat{A}") a la izquierda de la matriz o vector =mob.
PAISES = [tr("España", "Spain"), tr("Francia", "France"), tr("Alemania", "Germany"),
          tr("Reino Unido", "United Kingdom"), tr("EE. UU.", "USA")]
PIB = {2003: [2.9, 0.9, -0.2, 2.5, 1.8], 2004: [3.1, 2.0, 1.1, 3.2, 3.1],
       2005: [3.3, 1.6, 0.7, 1.9, 3.7], 2006: [3.3, 2.0, 0.9, 1.5, 3.5]}   # Sección 1.3 del libro (fuente: ICAE)
ANYOS = list(PIB)
FILAS_PIB = [[PIB[a][k] for a in ANYOS] for k in range(len(PAISES))]

A_EJ = [[4, -1, 0], [2, 2, 7]]      # transparencia "Matrices": a=(4,2,), b=(-1,2,), c=(0,7,)
COLOR_FILA, COLOR_COLUMNA = COLOR_A, COLOR_B   # índice de la fila y de la columna (letras del superíndice)


def vector(valores, color=WHITE, decimales=1, escala=0.8):
    conf = {"num_decimal_places": decimales, "group_with_commas": False}
    if decimales:
        v = DecimalMatrix([[x] for x in valores], left_bracket="(", right_bracket=")", v_buff=0.62, element_to_mobject_config=conf)
    else:
        v = IntegerMatrix([[x] for x in valores], left_bracket="(", right_bracket=")", v_buff=0.62)
    v.get_entries().set_color(color)
    return v.scale(escala)


def matriz(filas, color=WHITE, decimales=1, escala=0.8, h_buff=1.3):
    conf = {"num_decimal_places": decimales, "group_with_commas": False}
    if decimales:
        m = DecimalMatrix(filas, v_buff=0.62, h_buff=h_buff, element_to_mobject_config=conf)
    else:
        m = IntegerMatrix(filas, v_buff=0.62, h_buff=h_buff)
    m.get_entries().set_color(color)
    return m.scale(escala)


def con_nombre(tex, mob, font_size=48):
    return VGroup(MathTex(tex, tex_template=myTemplate, font_size=font_size), mob).arrange(RIGHT, buff=0.2)


def leyenda(rotulos, entradas, x_der, font_size=30):
    """Rótulos alineados a la derecha de x_der, cada uno a la altura de su entrada."""
    filas = VGroup(*[Tex(r, font_size=font_size) for r in rotulos])
    for f, e in zip(filas, entradas):
        f.move_to([x_der - f.width/2, e.get_y(), 0])
    return filas


def tabla_pib(centro=ORIGIN, font_size=30):
    """La matriz del PIB: [0] la matriz, [1] los países, [2] los años (encima de cada columna)."""
    M = matriz(FILAS_PIB).move_to(centro)
    paises = leyenda(PAISES, M.get_columns()[0], M.get_left()[0] - 0.3, font_size=font_size)
    anyos = VGroup(*[Tex(str(a), font_size=font_size, color=GRAY_B).next_to(col, UP, buff=0.35).set_y(M.get_top()[1] + 0.3)
                     for a, col in zip(ANYOS, M.get_columns())])
    return VGroup(M, paises, anyos)


def franja_fila(m, i, x0=None, x1=None, color=COLOR_ESCALAR):
    """Franja sobre la fila i (desde 0) de la matriz m, de x0 a x1 (por defecto, la matriz entera)."""
    x0 = m.get_left()[0] if x0 is None else x0
    x1 = m.get_right()[0] if x1 is None else x1
    y = m.get_rows()[i].get_y()
    return Rectangle(width=x1 - x0, height=0.45, stroke_width=0, fill_color=color,
                     fill_opacity=0.22).move_to([(x0 + x1)/2, y, 0])


def franja_columna(m, j, color=COLOR_ESCALAR):
    """Franja sobre la columna j (desde 0) de la matriz m."""
    col = m.get_columns()[j]
    return Rectangle(width=col.width + 0.35, height=m.get_brackets()[0].height, stroke_width=0, fill_color=color,
                     fill_opacity=0.22).move_to([col.get_x(), m.get_y(), 0])

1.6.3. Escena 2 - Ejemplos: la posición de cada vector importa

El primer ejemplo de V01, \((1,2,3)\neq(3,2,1)\), en versión matricial (transparencia "Matrices"): los tres vectores, escritos en vertical, pasan a ser las columnas de \(\Mat{A}\); con los vectores en otra posición la matriz es otra. 

self.creditos()
cabecera = Tex(tr("Ejemplos", "Examples"), font_size=44).set_color(COLOR_TITULO).to_edge(UP)
cols = list(zip(*A_EJ))   # las columnas de A: a, b y c
COLORES = [COLOR_A, COLOR_B, COLOR_C]   # un color por vector, para seguirlos cuando cambian de sitio
vs = [vector(c, color=k, decimales=0) for c, k in zip(cols, COLORES)]
vecs = VGroup(*[con_nombre(rf"\Vect{{{l}}}=", v) for l, v in zip("abc", vs)]).arrange(RIGHT, buff=1.0).move_to([0, 1.9, 0])
for v, k in zip(vecs, COLORES):
    v[0][0][0].set_color(k)   # la letra del vector
A = matriz(A_EJ, decimales=0, h_buff=1.0)
f_A = con_nombre(r"\Mat{A}=\begin{bmatrix}\Vect{a}; & \Vect{b}; & \Vect{c};\end{bmatrix}=", A).move_to([0, -0.1, 0])
ORDEN = (1, 2, 0)   # la nueva lista [b; c; a]: todos los vectores cambian de sitio
POS = {j: p for p, j in enumerate(ORDEN)}   # posición de cada vector en la nueva lista
C = matriz([[r[j] for j in ORDEN] for r in A_EJ], decimales=0, h_buff=1.0)
f_C = VGroup(MathTex(r"\begin{bmatrix}\Vect{b}; & \Vect{c}; & \Vect{a};\end{bmatrix}=", tex_template=myTemplate, font_size=48), C,
             MathTex(r"\neq\Mat{A}", tex_template=myTemplate, font_size=48)).arrange(RIGHT, buff=0.2).move_to([0, -2.4, 0])
f_C.shift((f_A[1].get_x() - C.get_x())*RIGHT)   # las dos matrices numéricas, una encima de la otra
# glifos: f_A[0] = A = [ a ; b ; c ; ] =  (letras 3, 5, 7);  f_C[0] = [ b ; c ; a ; ] =  (letras 1, 3, 5)
letras_A = [f_A[0][0][3], f_A[0][0][5], f_A[0][0][7]]
letras_C = [f_C[0][0][1 + 2*POS[j]] for j in range(3)]   # dónde acaba cada vector (a, b, c)
for k, la, lc in zip(COLORES, letras_A, letras_C):
    la.set_color(k); lc.set_color(k)
for j, k in enumerate(COLORES):
    A.get_columns()[j].set_color(k)
    C.get_columns()[POS[j]].set_color(k)
resto_C = VGroup(*[g for g in f_C[0][0] if all(g is not l for l in letras_C)], C.get_brackets(), f_C[2])

with self.voiceover(text=tr("""Veamos un ejemplo con tres vectores de dos componentes.""",
                            """Let us look at an example with three vectors of two components.""")) as tracker:
    self.play(Write(cabecera), LaggedStart(*[FadeIn(v) for v in vecs], lag_ratio=0.4), run_time=1.5)

with self.voiceover(text=tr("""Con ellos formamos la matriz A, <bookmark mark='l'/> que es la lista de los tres vectores entre corchetes. <bookmark mark='c'/> Cada vector es una columna, y las columnas se escriben una al lado de la otra.""",
                            """With them we build the matrix A, <bookmark mark='l'/> which is the list of the three vectors in square brackets. <bookmark mark='c'/> Each vector is a column, and the columns are written side by side.""")) as tracker:
    self.wait_until_bookmark("l")
    self.play(Write(f_A[0]), run_time=1.2)
    self.wait_until_bookmark("c")
    self.play(FadeIn(A.get_brackets()), run_time=0.4)
    self.play(LaggedStart(*[TransformFromCopy(v.get_entries(), col) for v, col in zip(vs, A.get_columns())], lag_ratio=0.5), run_time=2)

with self.voiceover(text=tr("""Como en los vectores, la posición importa: <bookmark mark='o'/> si colocamos los vectores en otro lugar de la lista, obtenemos una matriz distinta.""",
                            """As with vectors, position matters: <bookmark mark='o'/> if we place the vectors elsewhere in the list, we get a different matrix.""")) as tracker:
    self.wait_until_bookmark("o")
    self.play(FadeIn(resto_C), run_time=0.6)
    for j in ORDEN:   # b, c y a, por el orden de la nueva lista: copias de las columnas de A
        fr = franja_columna(A, j)
        self.play(FadeIn(fr), run_time=0.3)
        self.play(TransformFromCopy(letras_A[j], letras_C[j]), TransformFromCopy(A.get_columns()[j], C.get_columns()[POS[j]]), run_time=0.9)
        self.play(FadeOut(fr), run_time=0.2)

self.pausa_media()

1.6.4. Escena 3 - Listas con distinto número de vectores: el orden

El segundo ejemplo de V01 (\(\R[4]\) frente a \(\R[2]\)) en versión matricial. \(\Mat{B}\) se forma con dos vectores que llevan los números de las filas de \(\Mat{A}\) (semilla de la transposición). Omega cree que son la misma matriz (mismos números); son listas distintas de vectores (tres de dos números frente a dos de tres, como dos vectores con listas distintas), y eso es lo que dice el orden (Marcos, 2026-10-18). Al final, como "pertenece a \(\R[4]\) por ser una lista de cuatro números": \(\Mat{A}\in\R[2\times 3]\) y \(\Mat{B}\in\R[3\times 2]\). Al decir que son listas distintas, cada matriz se escribe como la lista de sus columnas, entre paréntesis y seguidas de ";", como en el libro (Marcos, 2026-10-18), y vuelve a su forma de tabla para hablar del orden. 

self.creditos()
cabecera = Tex(tr("El orden de una matriz", "The order of a matrix"), font_size=44).set_color(COLOR_TITULO).to_edge(UP)
A = matriz(A_EJ, decimales=0, h_buff=1.0)
f_A = con_nombre(r"\Mat{A}=", A).move_to([-4.2, 0.9, 0])
ll_f = Brace(A, RIGHT, buff=0.1)
ll_c = Brace(A, UP, buff=0.15)
r_f = ll_f.get_tex(tr(r"2\ \text{filas}", r"2\ \text{rows}"), buff=0.1).scale(0.8)
r_c = ll_c.get_tex(tr(r"3\ \text{columnas}", r"3\ \text{columns}"), buff=0.1).scale(0.8)
B = matriz([list(f) for f in zip(*A_EJ)], decimales=0, h_buff=1.0)
f_B = con_nombre(r"\Mat{B}=", B).move_to([2.0, 0.9, 0])   # deja sitio a A escrita como lista
llB_f = Brace(B, RIGHT, buff=0.1)
llB_c = Brace(B, UP, buff=0.15)
rB_f = llB_f.get_tex(tr(r"3\ \text{filas}", r"3\ \text{rows}"), buff=0.1).scale(0.8)
rB_c = llB_c.get_tex(tr(r"2\ \text{columnas}", r"2\ \text{columns}"), buff=0.1).scale(0.8)
dimA = MathTex(r"\Matdim{A}{2}{3}", tex_template=myTemplate, font_size=56)

3.2. Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la clase del vídeo completo): corregir una escena no obliga a renderizarlo todo ni a pagar otra vez la voz, y los límites entre escenas quedan exactos. Cada segmento se recodifica igualando la duración del audio y del vídeo (concatenar sin recodificar deja un parón en cada unión, pues manim genera el audio unas décimas más largo que la imagen). La música suena en la portada (volumen volumen) y se funde cuando empieza la primera escena; vuelve al empezar la escena cierre (bajo la voz, con volumen_voz), sube en la última escena (los créditos) y se funde a negro con la imagen. La voz se mide entera y se lleva a -16 LUFS con una ganancia fija y un limitador (no con loudnorm en una pasada, que subía el volumen poco a poco al empezar). Los .srt de las escenas se desplazan y se funden en un único video.srt junto al mp4 (para subirlo a YouTube).

escenas es la lista ordenada de clases; cierre la escena en la que vuelve la música (el resumen). El mp4 y el srt finales quedan en la raíz del repositorio del idioma. 

export LC_ALL=C   # decimales con punto para bc y printf
[ "$idioma" = "en" ] && raiz=../videoMath2 || raiz=.
dir=$raiz/lib/media/videos/$fichero/$calidad
tmp=$raiz/tmp/$video; mkdir -p $tmp; rm -f $tmp/seg_*.mp4

# 1. Un segmento por escena, con audio y vídeo de la misma duración exacta
i=0; lista=""; offsets=""; T=0; T0=""; Tc=""
for esc in $escenas; do
  seg=$(printf "%s/seg_%02d.mp4" $tmp $i)
  dur=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=duration -of default=nk=1:nw=1 $dir/$esc.mp4)
  fps=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=r_frame_rate -of default=nk=1:nw=1 $dir/$esc.mp4)
  if ffprobe -loglevel error -select_streams a -show_entries stream=codec_type -of csv=p=0 $dir/$esc.mp4 | grep -q audio; then
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -af "aresample=48000,apad" -ac 2 -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  else
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -f lavfi -i anullsrc=r=48000:cl=stereo -map 0:v -map 1:a -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  fi
  d=$(ffprobe -loglevel error -show_entries format=duration -of default=nk=1:nw=1 $seg)
  offsets="$offsets $T"; [ "$esc" = "$cierre" ] && Tc=$T
  printf "%2d  %-55s %8.3f s  (empieza en %8.3f)\n" $i $esc $d $T
  T=$(bc -l <<< "$T+$d"); [ -z "$T0" ] && T0=$d; dl=$d
  lista="$lista $seg"; i=$((i+1))
done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
# La voz se mide entera y recibe una ganancia fija hasta -16 LUFS, con un limitador para los picos.
# (loudnorm en una pasada ajusta la ganancia sobre la marcha: la primera frase quedaba 5 dB más baja.)
ia=""; for k in $(seq 0 $((n-1))); do ia="$ia[$k:a]"; done
I=$(ffmpeg -hide_banner -nostats $inputs -filter_complex "${ia}concat=n=$n:v=0:a=1,ebur128=framelog=quiet" -f null - 2>&1 | awk '$1=="I:"{v=$2} END{print v}')
ganancia=$(bc -l <<< "-16 - ($I)"); printf "Voz: %s LUFS; ganancia %.2f dB\n" $I $ganancia
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]volume=${ganancia}dB,alimiter=limit=0.84:level=false[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];
  [$((n+2)):a]atrim=$dur_m2:$fin_m3,asetpts=PTS-STARTPTS,volume=$volumen,afade=t=in:st=0:d=1,adelay=delays=${ms_m3}:all=1[m3];
  [voz][m1][m2][m3]amix=inputs=4:duration=first:normalize=0[mix];
  [mix]afade=t=out:st=$tfade:d=6[a];
  [v0]fade=t=out:st=$tfade:d=6[v]" \
  -map "[v]" -map "[a]" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -c:a aac -b:a 192k $raiz/$video.mp4
echo "Vídeo: $raiz/$video.mp4"

import sys, os, re
dir, salida = sys.argv[1], sys.argv[2]
escenas = os.environ["ESCENAS"].split(); offsets = [float(x) for x in os.environ["OFFSETS"].split()]
def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000
def s2t(x):
    ms = round(x*1000); h, r = divmod(ms, 3600000); m, r = divmod(r, 60000); s, ms = divmod(r, 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"
k, out = 1, []
for esc, off in zip(escenas, offsets):
    ruta = f"{dir}/{esc}.srt"
    if not os.path.exists(ruta): continue
    for bloque in re.split(r"\n\s*\n", open(ruta, encoding="utf-8").read().strip()):
        lineas = bloque.strip().splitlines()
        if len(lineas) < 3 or "-->" not in lineas[1]: continue
        a, _, b = lineas[1].partition("-->")
        out.append(f"{k}\n{s2t(t2s(a)+off)} --> {s2t(t2s(b)+off)}\n" + "\n".join(lineas[2:])); k += 1
open(salida, "w", encoding="utf-8").write("\n\n".join(out) + "\n")
print(f"Subtítulos: {salida} ({k-1} entradas)")
EOF

Recorre el fichero de python y saca, escena por escena, los textos tr(...) del idioma pedido: en prosa la narración (los que están dentro de self.voiceover) y entre corchetes los textos que aparecen en pantalla. import ast, textwrap

arbol = ast.parse(open(f"lib/{fichero}.py", encoding="utf-8").read())
idx   = 0 if idioma == "es" else 1

def texto(nodo):
    try:
        return ast.literal_eval(nodo)
    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
