def matriz(filas, color=WHITE, escala=0.8, h_buff=1.0):
    m = IntegerMatrix(filas, v_buff=0.62, h_buff=h_buff)
    m.get_entries().set_color(color)
    return m.scale(escala)


def con_nombre(tex, mob, font_size=48):
    return VGroup(MathTex(tex, tex_template=myTemplate, font_size=font_size), mob).arrange(RIGHT, buff=0.2)


def diagonal(m):
    """Las componentes de la diagonal principal de la matriz m (también si es rectangular)."""
    filas = m.get_rows()
    return [filas[k][k] for k in range(min(len(filas), len(filas[0])))]


def resalta(escena, mobs, color=COLOR_ESCALAR, factor=1.3, run_time=0.3):
    """Resaltado inmediato y sostenido (color y tamaño), sin la subida lenta de Indicate."""
    escena.play(*[x.animate.set_color(color).scale(factor) for x in mobs], run_time=run_time)


def apaga(escena, mobs, color=WHITE, factor=1.3, run_time=0.3):
    escena.play(*[x.animate.set_color(color).scale(1/factor) for x in mobs], run_time=run_time)


def columna_de_lineas(lineas, x0, y0, dy):
    """Fórmulas de varias partes una debajo de otra, con la parte 1 (el "=") alineada en x0."""
    for k, l in enumerate(lineas):
        l.shift([x0 - l[1].get_left()[0], y0 - k*dy - l.get_y(), 0])
    return VGroup(*lineas)


def copia_literal(escena, origen, destino, igual, run_time=1.2):
    """Copia la expresión origen tal cual, alineada a la izquierda donde irá destino, y escribe el igual."""
    copia = origen.copy()
    copia.shift((destino.get_left()[0] - copia.get_left()[0])*RIGHT)
    escena.play(FadeIn(igual), TransformFromCopy(origen, copia), run_time=run_time)
    return copia


def transforma_copia(escena, copia, destino, parejas, se_van=(), aparecen=(), run_time=1.5):
    """Sobre la copia: los glifos de se_van desaparecen, los de parejas (glifo de la copia, glifo de
    destino) van a su sitio, y los de aparecen (índices de destino) surgen donde van."""
    escena.play(*[FadeOut(copia[i]) for i in se_van],
                *[ReplacementTransform(copia[i], destino[j]) for i, j in parejas],
                *[FadeIn(destino[j]) for j in aparecen], run_time=run_time)


def construye_transpuesta(escena, M, MT, nombreT=None, run_time=0.6):
    """Ejemplo de transposición: cada fila de M viaja a la columna del mismo índice de MT, una a una."""
    escena.play(*([FadeIn(nombreT)] if nombreT is not None else []), FadeIn(MT.get_brackets()), run_time=0.5)
    filas, cols = M.get_rows(), MT.get_columns()
    for r in range(len(filas)):
        escena.play(*[TransformFromCopy(filas[r][c], cols[r][c]) for c in range(len(filas[r]))], run_time=run_time)


def rotulo(texto, junto_a, x=None, font_size=28):
    """Rótulo gris pequeño a la derecha de una línea (la regla que justifica el paso)."""
    r = Tex(texto, tex_template=myTemplate, font_size=font_size, color=GRAY_B)
    r.next_to(junto_a, RIGHT, buff=0.6)
    if x is not None:
        r.shift((x - r.get_left()[0])*RIGHT)
    return r

class Panel:
    """Panel de herramientas como el de V06-V07 (copia del PanelV10 de V11), con una pieza a cada lado:
    [(cabecera, clave, fórmula), ...]. Pista (amarillo y recuadro), parpadeo antes de cada paso y
    contador de usos con campanita."""
    def __init__(self, escena, piezas, y_sup=2.95, font_size=36):
        self.escena, self.piezas, self.usos, self.contadores = escena, {}, {}, {}
        self.marcos = VGroup()
        lados = []
        for cab, clave, formula in piezas:
            m = MathTex(*formula, tex_template=myTemplate, font_size=font_size) if isinstance(formula, (list, tuple)) \
                else MathTex(formula, tex_template=myTemplate, font_size=font_size)
            self.piezas[clave] = m
            lados.append(VGroup(Tex(cab, font_size=24).set_color(GRAY_B), m).arrange(DOWN, aligned_edge=LEFT, buff=0.2))
        lados[0].to_edge(LEFT, buff=0.5)
        lados[1].to_edge(RIGHT, buff=1.6)   # deja sitio al contador sin tocar la firma del margen
        for g in lados:
            g.align_to([0, y_sup, 0], UP)
        self.todo = VGroup(*lados)

    def pista(self, *claves, run_time=1):
        marcos = VGroup(*[SurroundingRectangle(self.piezas[c], color=COLOR_ESCALAR, buff=0.1,
                                               corner_radius=0.1, stroke_width=2) for c in claves])
        self.marcos.add(*marcos)
        self.escena.play(*[self.piezas[c].animate.set_color(COLOR_ESCALAR) for c in claves],
                         Create(marcos), run_time=run_time)

    def senala(self, clave, run_time=0.8):
        self.escena.play(Indicate(self.piezas[clave], color=COLOR_RESULTADO, scale_factor=1.12), run_time=run_time)

    def cuenta(self, clave, run_time=0.5):
        self.usos[clave] = self.usos.get(clave, 0) + 1
        disco = Circle(radius=0.17, fill_color=COLOR_TITULO, fill_opacity=1, stroke_width=0)
        numero = Tex(f"\\textbf{{{self.usos[clave]}}}", font_size=24).set_color(WHITE).move_to(disco)
        nuevo = VGroup(disco, numero).next_to(self.piezas[clave], RIGHT, buff=0.3).shift(0.1*UP)
        self.escena.add_sound(CAMPANITA)
        if clave in self.contadores:
            self.escena.play(ReplacementTransform(self.contadores[clave], nuevo), run_time=run_time)
        else:
            self.escena.play(FadeIn(nuevo, scale=0.5), run_time=run_time)
        self.contadores[clave] = nuevo

** Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la
clase del vídeo completo): corregir una escena no obliga a renderizarlo
todo ni a pagar otra vez la voz, y los límites entre escenas quedan
exactos. Cada segmento se recodifica igualando la duración del audio y
del vídeo (concatenar sin recodificar deja un parón en cada unión, pues
manim genera el audio unas décimas más largo que la imagen). La música
suena en la portada (volumen =volumen=) y se funde cuando empieza la
primera escena; vuelve al empezar la escena =cierre= (bajo la voz, con
=volumen_voz=), sube en la última escena (los créditos) y se funde a
negro con la imagen. La voz se mide entera y se lleva a -16 LUFS con
una ganancia fija y un limitador (no con =loudnorm= en una pasada, que
subía el volumen poco a poco al empezar). Los =.srt= de las escenas se desplazan y se
funden en un único =video.srt= junto al mp4 (para subirlo a YouTube).

=escenas= es la lista ordenada de clases; =cierre= la escena en la que
vuelve la música (el resumen). El mp4 y el srt finales quedan en la raíz del repositorio del
idioma.

done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
# La voz se mide entera y recibe una ganancia fija hasta -16 LUFS, con un limitador para los picos.
# (loudnorm en una pasada ajusta la ganancia sobre la marcha: la primera frase quedaba 5 dB más baja.)
ia=""; for k in $(seq 0 $((n-1))); do ia="$ia[$k:a]"; done
I=$(ffmpeg -hide_banner -nostats $inputs -filter_complex "${ia}concat=n=$n:v=0:a=1,ebur128=framelog=quiet" -f null - 2>&1 | awk '$1=="I:"{v=$2} END{print v}')
ganancia=$(bc -l <<< "-16 - ($I)"); printf "Voz: %s LUFS; ganancia %.2f dB\n" $I $ganancia
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]volume=${ganancia}dB,alimiter=limit=0.84:level=false[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];
  [$((n+2)):a]atrim=$dur_m2:$fin_m3,asetpts=PTS-STARTPTS,volume=$volumen,afade=t=in:st=0:d=1,adelay=delays=${ms_m3}:all=1[m3];
  [voz][m1][m2][m3]amix=inputs=4:duration=first:normalize=0[mix];
  [mix]afade=t=out:st=$tfade:d=6[a];
  [v0]fade=t=out:st=$tfade:d=6[v]" \
  -map "[v]" -map "[a]" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -c:a aac -b:a 192k $raiz/$video.mp4
echo "Vídeo: $raiz/$video.mp4"

# 3. Subtítulos: desplazar el srt de cada escena y fundirlos en uno
ESCENAS="$escenas" OFFSETS="$offsets" python3 - "$dir" "$raiz/$video.srt" <<'EOF'
import sys, os, re
dir, salida = sys.argv[1], sys.argv[2]
escenas = os.environ["ESCENAS"].split(); offsets = [float(x) for x in os.environ["OFFSETS"].split()]
def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000
def s2t(x):
    ms = round(x*1000); h, r = divmod(ms, 3600000); m, r = divmod(r, 60000); s, ms = divmod(r, 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"
k, out = 1, []
for esc, off in zip(escenas, offsets):
    ruta = f"{dir}/{esc}.srt"
    if not os.path.exists(ruta): continue
    for bloque in re.split(r"\n\s*\n", open(ruta, encoding="utf-8").read().strip()):
        lineas = bloque.strip().splitlines()
        if len(lineas) < 3 or "-->" not in lineas[1]: continue
        a, _, b = lineas[1].partition("-->")
        out.append(f"{k}\n{s2t(t2s(a)+off)} --> {s2t(t2s(b)+off)}\n" + "\n".join(lineas[2:])); k += 1
open(salida, "w", encoding="utf-8").write("\n\n".join(out) + "\n")
print(f"Subtítulos: {salida} ({k-1} entradas)")
EOF

** Extraer el guion (la narración) para revisarlo

Recorre el fichero de python y saca, escena por escena, los textos
~tr(...)~ del idioma pedido: en prosa la narración (los que están dentro
de ~self.voiceover~) y entre corchetes los textos que aparecen en
pantalla. Sirve para revisar el guion sin leer código y para la
descripción o los subtítulos del vídeo. Como los dos py solo difieren
en =LANG=, lee siempre el fichero español (=./lib/=).

#+name: guion
#+BEGIN_SRC python :python python3 :var fichero="LXX_VYY_FileName" :var idioma="es" :results output
import ast, textwrap

arbol = ast.parse(open(f"lib/{fichero}.py", encoding="utf-8").read())
idx   = 0 if idioma == "es" else 1

def texto(nodo):
    try:
        return ast.literal_eval(nodo)
    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
#+END_SRC
