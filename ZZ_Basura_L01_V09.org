** Escena 6 - La igualdad de matrices

Dos matrices del mismo orden; una franja recorre las columnas de las dos
a la vez mientras la voz las empareja; debajo, el criterio con el
selector.

#+name: Sexta escena
#+BEGIN_SRC python :eval no
self.creditos()
A = matriz([[1, 1, 9], [2, 0, 2]], decimales=0, h_buff=1.0)
B = matriz([[1, 1, 9], [2, 0, 2]], decimales=0, h_buff=1.0)
par = VGroup(VGroup(MathTex(r"\Mat{A}=", tex_template=myTemplate, font_size=48), A).arrange(RIGHT, buff=0.2),
             VGroup(MathTex(r"\Mat{B}=", tex_template=myTemplate, font_size=48), B).arrange(RIGHT, buff=0.2)).arrange(RIGHT, buff=1.5).move_to([0, 1, 0])
criterio = MathTex(r"\Mat{A}=\Mat{B}", r"\iff", r"\VectC{A}{j}=\VectC{B}{j}", tr(r"\quad\text{para } j=1:n", r"\quad\text{for } j=1:n"),
                   tex_template=myTemplate, font_size=48).move_to([0, -1.6, 0])
cabecera = Tex(tr("Igualdad de matrices", "Equality of matrices"), font_size=44).set_color(COLOR_TITULO).to_edge(UP)

with self.voiceover(text=tr("""Dos matrices del mismo orden son iguales cuando lo son sus columnas: <bookmark mark='c1'/> la primera con la primera, <bookmark mark='c2'/> la segunda con la segunda, <bookmark mark='c3'/> y así con todas.""",
                            """Two matrices of the same order are equal when their columns are: <bookmark mark='c1'/> the first with the first, <bookmark mark='c2'/> the second with the second, <bookmark mark='c3'/> and so on with all of them.""")) as tracker:
    self.play(Write(cabecera), FadeIn(par), run_time=1)
    franjas = None
    for j, marca in enumerate(["c1", "c2", "c3"]):
        self.wait_until_bookmark(marca)
        nuevas = VGroup(franja_columna(A, j), franja_columna(B, j))
        self.play(FadeIn(nuevas) if franjas is None else ReplacementTransform(franjas, nuevas), run_time=0.6)
        franjas = nuevas

with self.voiceover(text=tr("""Es decir, columna a columna. <bookmark mark='r'/> Este mismo criterio nos servirá para definir las operaciones con matrices.""",
                            """That is, column by column. <bookmark mark='r'/> This same criterion will help us define the operations with matrices.""")) as tracker:
    self.play(FadeOut(franjas), Write(criterio), run_time=1.5)
    self.wait_until_bookmark("r")
    self.play(Indicate(criterio[2], color=COLOR_ESCALAR, scale_factor=1.1), run_time=1)

self.pausa_media()
