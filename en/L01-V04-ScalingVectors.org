(defvar my/src-block-names nil)

(defun my/build-src-block-names (file)
  "Construye una lista de (posicion . nombre) para bloques con #+NAME."
  (setq my/src-block-names nil)
  (with-temp-buffer
    (insert-file-contents file)
    (goto-char (point-min))
    (while (re-search-forward
            "^#\\+NAME:[ \t]*\\(.+\\)\n.*#\\+BEGIN_SRC" nil t)
      (push (cons (match-beginning 0)
                  (string-trim (match-string 1)))
            my/src-block-names)))
  (setq my/src-block-names (nreverse my/src-block-names)))

(defun my/org-html-src-block-with-name (src-block contents info)
  (let* ((begin (org-element-property :begin src-block))
         ;; Buscar el nombre cuya posición esté justo antes de este bloque
         ;; y sea el más cercano
         (label
          (let ((best nil)
                (best-pos -1))
            (dolist (entry my/src-block-names)
              (let ((pos (car entry))
                    (name (cdr entry)))
                ;; La posición en el fichero original es aproximada;
                ;; buscamos la entrada cuya pos sea <= begin y la más cercana
                (when (and (<= pos begin)
                           (> pos best-pos))
                  (setq best name)
                  (setq best-pos pos))))
            ;; Solo usar si está suficientemente cerca (500 chars)
            (when (and best (< (- begin best-pos) 500))
              best)))
         (result (org-html-src-block src-block contents info)))
    (if (and label (not (string-empty-p label)))
        (concat
         (format "<div class=\"src-block-name\"><code>%s</code></div>\n" label)
         result)
      result)))

(defun my/html-export-to-file (&optional async subtreep visible-only body-only ext-plist)

def retículaProducto(a, b: int, ax: Axes) -> VGroup:
    def cuadricula(a: int, b: int, ax: Axes):
        base = VGroup(*[Square().scale(0.5) for s in range(0,int(a*b))])
        base.arrange_in_grid(rows=b,
                             flow_order="ru",
                             buff=0).next_to(ax.coords_to_point(0, 0),
                                             UR,
                                             buff=0)
        return Group(*[base[c[0]:c[1]] for c in [(a*i,a*(i+1)) for i in range(b)]])
    
    def rectangulo(a, b: int, ax: Axes):
        base  = VGroup(*[Rectangle(height=2.0, width=2.0*a).scale(0.5) for s in range(0,b)])
        posicion = UR if a>0 else UL
        tabla = base.arrange_in_grid(rows=b,
                                     flow_order="ru",
                                     buff=0).next_to(ax.coords_to_point(0, 0),
                                                     posicion,
                                                     buff=0)
        return tabla
    return cuadricula(a, b, ax) if isinstance(a, int) else rectangulo(a, b, ax)

def multiplica(a, b, ax: Axes):
    """
    Desplaza horizontalmente una recta que pasa por (a, 0) y (0, 1)
    hasta que corte el eje vertical en el punto (0, b)
    """
    # Diagonal del rectángulo
    punto_inferior = ax.coords_to_point(a, 0)
    punto_superior = ax. coords_to_point(0, 1)
    diagonal = Line(punto_inferior, punto_superior, color=BLUE)
    # Extender la diagonal.
    # Con la ecuación: y = 1 - x/a, podemos calcular puntos más alejados
    if b>0:
        y_arriba = 1+b # Punto lejano a la izquierda
        x_arriba = -a*(y_arriba-1)
        y_abajo = -1  # Punto lejano a la derecha
        x_abajo = 2*a
    else:
        y_arriba = 2 # Punto lejano a la izquierda
        x_arriba = -a
        y_abajo = b-1  # Punto lejano a la derecha
        x_abajo = -a*(y_abajo-1)
        # Diagonal extendida para cortar los ejes tras el desplazamiento
    diagonal_extendida = Line(
        ax.coords_to_point(x_arriba, y_arriba),
        ax.coords_to_point(x_abajo, y_abajo),
        color=RED                )                
    desplazamiento_x = a * (b - 1)
    desplazamiento = ax.coords_to_point(desplazamiento_x, 0) - ax.coords_to_point(0, 0)
    diagonal_extendida.shift(desplazamiento)
    
    return diagonal, diagonal_extendida

def line_intersection(l1, l2):
    """
    Devuelve el punto de intersección entre dos rectas dadas
    por pares de puntos (p1, p2), (p3, p4).
    """
    p1, p2 = l1
    p3, p4 = l2
    x1, y1 = p1[:2]
    x2, y2 = p2[:2]
    x3, y3 = p3[:2]
    x4, y4 = p4[:2]
    denom = (x1-x2)*(y3-y4) - (y1-y2)*(x3-x4)
    px = ((x1*y2-y1*x2)*(x3-x4)-(x1-x2)*(x3*y4-y3*x4))/denom
    py = ((x1*y2-y1*x2)*(y3-y4)-(y1-y2)*(x3*y4-y3*x4))/denom
    return np.array([px, py, 0])

def multiplica(vector, c, axes: Axes, unidad):
    """
    Multiplica un vector de R2 por un escalar
    """
    def diagonales(x,y,c=1):
        diagonalx = Line(axes.coords_to_point(x*c, 0),
                         axes.coords_to_point(0, 1*c),
                         color=BLUE)
        diagonaly = Line(axes.coords_to_point(1*c, 0),
                         axes.coords_to_point(0, y*c),
                         color=BLUE)
        return diagonalx, diagonaly
        
    x,y,z = vector.coords
    # arcos (el segundo marca |c|)
    arco1 = Arco(axes, unidad)
    arco2 = Arco(axes, unidad*c)
    # Diagonales de x y de y
    diagonalx1, diagonaly1 = diagonales(x,y)
    diagonalx2, diagonaly2 = diagonales(x,y,c)
    return arco1, diagonalx1, diagonaly1, arco2, diagonalx2, diagonaly2

def Arcos(axes, radio):
    # Obtener la posición del punto (0,0,0) en los ejes
    origen_ejes = axes.c2p(0, 0, 0)
    # Radio en coordenadas de los ejes
    radio_coordenadas = radio
    # Convertir el radio a unidades de Manim
    radio_manim = np.linalg.norm(
        axes.c2p(radio_coordenadas, 0, 0) - axes.c2p(0, 0, 0))
    # si radio negativo nos vamos al cuadrante negativo con start_angle=PI
    signo = 0 if radio>0 else PI
    # Arco entre X e Y (en el plano XY)
    arco_xy = Arc(
        radius=radio_manim,
        start_angle=signo,
        angle=PI/2,
        arc_center=origen_ejes,
        color=YELLOW,
        stroke_width=3)
    # Arco entre Y y Z (en el plano YZ)
    arco_yz = Arc(
        radius=radio_manim,
        start_angle=signo,
        angle=PI/2,
        arc_center=origen_ejes,
        color=YELLOW,
        stroke_width=3).rotate(-PI/2, axis=UP, about_point=origen_ejes)
    # Arco entre Z y X (en el plano ZX)
    arco_zx = Arc(
        radius=radio_manim,
        start_angle=signo,
        angle=PI/2,
        arc_center=origen_ejes,
        color=YELLOW,
        stroke_width=3).rotate(PI/2, axis=RIGHT, about_point=origen_ejes)

    arcos = VGroup(arco_xy, arco_yz, arco_zx)
    return arcos      

def coordenadasYdiagonales(tupla, axes, factor=1):
    
    coordenadas = tupla
    Vector = nc.Vector(coordenadas)

    Vx = nc.Vector([coordenadas[0],0,0,])
    Vy = nc.Vector([0,coordenadas[1],0,])
    Vz = nc.Vector([0,0,coordenadas[2],])
    Vxy = nc.Vector([coordenadas[0],coordenadas[1],0,])

    # líneas de las coordenadas
    line_x = Line(start=np.array(axes.c2p(*Vx.lista)),
                  end=np.array(axes.c2p(*Vxy.lista)),
                  stroke_width=1.0)
    line_y = Line(start=np.array(axes.c2p(*Vy.lista)),
                  end=np.array(axes.c2p(*Vxy.lista)),
                  stroke_width=1.0)
    line_xy = Line(start=np.array(axes.c2p(*Vxy.lista)),
                   end=np.array(axes.c2p(*Vector.lista)),
                   stroke_width=1.0)
    line_z = Line(start=np.array(axes.c2p(*Vz.lista)),
                  end=np.array(axes.c2p(*Vector.lista)),
                  stroke_width=1.0)
    
    lineas = VGroup(line_x, line_y, line_xy, line_z)

    # Diagonales
    diagonal_x = Line3D(
        start=axes.c2p(*[factor if x != 0 else 0 for x in (0,1,0)]),
        end=axes.c2p(*Vx.lista),
        #stroke_width=1.9,
        color=RED)

    diagonal_y = Line3D(
        start=axes.c2p(*[factor if x != 0 else 0 for x in (0,0,1)]),
        end=axes.c2p(*Vy.lista),
        #stroke_width=1.9,
        color=GREEN)

    diagonal_z = Line3D(
        start=axes.c2p(*[factor if x != 0 else 0 for x in (1,0,0)]),
        end=axes.c2p(*Vz.lista),
        #stroke_width=1.9,
        color=BLUE)

    diagonales = VGroup(diagonal_x, diagonal_y, diagonal_z)

    return lineas, diagonales

class VectorR3():
    def __init__(self, lista, rpr='c', color=GRAY_B):
        """Inicializa un Vector con una lista"""
        coords = lista + [0] if len(lista)<3 else lista
        self.color  = color
        #self.coords = tuple(coords)
        self.coords = tuple([int(i) if float(i)==int(i) else float(i) for i in coords])
        self.Vector = nc.Vector(self.coords, rpr)
        self.tex    = MathTex(self.Vector.latex(), color=self.color).scale(0.8)
        self.proyXY = self.coords[:-1] + (0,)
        self.proyZ  = (0,0,) + (self.coords[-1],)

    def dot(self, ejes, radio=0.08):
        return Dot3D(ejes.c2p(*self.coords), radius=radio, color=self.color)

    def x_line(self, ejes):
        return ejes.get_vertical_line(ejes.c2p(*self.proyXY), color=self.color)
    
    def y_line(self, ejes):
        return ejes.get_horizontal_line(ejes.c2p(*self.proyXY), color=self.color)
    
    def z_line(self, ejes):
        return Line3D(start=ejes.c2p(*self.proyZ), end=ejes.c2p(*self.coords), thickness=0.01)
    
    def v_line(self, ejes):
        return Line3D(start=ejes.c2p(*self.proyXY), end=ejes.c2p(*self.coords), thickness=0.01)
    
    def xy_line(self, ejes):
        return Line3D(start=np.array([0,0,0]), end=ejes.c2p(*self.proyXY), thickness=0.01)
    
    def proy_x_line(self, ejes):
        return ejes.get_horizontal_line(ejes.c2p(*self.proyXY), color=self.color)
    
    def proy_y_line(self, ejes):
        return ejes.get_horizontal_line(ejes.c2p(*self.coords), color=self.color)

    def proy_y_line(self, ejes):
        return Line3D(start=ejes.c2p(*self.proyZ), end=ejes.c2p(*self.coords), thickness=0.01)

def multiplica(a, b, ax: Axes):
    """
    Desplaza horizontalmente una recta que pasa por (a, 0) y (0, 1)
    hasta que corte el eje vertical en el punto (0, b)
    """
    # Diagonal del rectángulo
    punto_inferior = ax.coords_to_point(a, 0)
    punto_superior = ax. coords_to_point(0, 1)
    diagonal = Line(punto_inferior, punto_superior, color=BLUE)
    # Extender la diagonal
    if b>0:
        y_arriba = 1+b
        x_arriba = -a*(y_arriba-1)
        y_abajo = -1
        x_abajo = 2*a
    else:
        y_arriba = 2
        x_arriba = -a
        y_abajo = b-1
        x_abajo = -a*(y_abajo-1)
    diagonal_extendida = Line(
        ax.coords_to_point(x_arriba, y_arriba),
        ax.coords_to_point(x_abajo, y_abajo),
        color=RED
    )                
    desplazamiento_x = a * (b - 1)
    desplazamiento = ax.coords_to_point(desplazamiento_x, 0) - ax.coords_to_point(0, 0)
    diagonal_extendida.shift(desplazamiento)
    return diagonal, diagonal_extendida

def line_intersection(l1, l2):
    """
    Devuelve el punto de intersección entre dos rectas dadas
    por pares de puntos (p1, p2), (p3, p4).
    """
    p1, p2 = l1
    p3, p4 = l2
    x1, y1 = p1[:2]
    x2, y2 = p2[:2]
    x3, y3 = p3[:2]
    x4, y4 = p4[:2]
    denom = (x1-x2)*(y3-y4) - (y1-y2)*(x3-x4)
    px = ((x1*y2-y1*x2)*(x3-x4)-(x1-x2)*(x3*y4-y3*x4))/denom
    py = ((x1*y2-y1*x2)*(y3-y4)-(y1-y2)*(x3*y4-y3*x4))/denom
    return np.array([px, py, 0])
