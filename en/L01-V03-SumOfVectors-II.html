Defining Vectors in R2 and R3

These classes define vectors in \(\R[2]\) and \(\R[3]\) to be represented with drawings. 

class VectorR2():
    def __init__(self, list, rpr='c', color=GRAY_B):
        """Initializes a Vector with a list"""
        coords = list + [0] if len(list) < 3 else list        
        self.color = color
        #self.coords = tuple(coords)
        #self.coords = tuple([float(i) for i in coords])
        self.coords = tuple([int(i) if float(i) == int(i) else float(i) for i in coords])
        self.Vector = nc.Vector(self.coords[:2], rpr)
        self.tex = MathTex(self.Vector.latex(), color=self.color).scale(0.8)

    def dot(self, axes, radius=0.08):
        return Dot(axes.c2p(*self.coords), radius=radius, color=self.color)

    def v_line(self, axes):
        return axes.get_vertical_line(axes.c2p(*self.coords), color=self.color)

    def h_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.coords), color=self.color)

    def arrow(self, axes):
        return axes.get_vector(self.Vector.lista, stroke_color=self.color, stroke_width=4)


class VectorR3():
    def __init__(self, list, rpr='c', color=GRAY_B):
        """Initializes a Vector with a list"""
        coords = list + [0] if len(list) < 3 else list
        self.color = color
        #self.coords = tuple(coords)
        self.coords = tuple([int(i) if float(i) == int(i) else float(i) for i in coords])
        self.Vector = nc.Vector(self.coords, rpr)
        self.tex = MathTex(self.Vector.latex(), color=self.color).scale(0.8)
        self.projXY = self.coords[:-1] + (0,)
        self.projZ  = (0, 0,) + (self.coords[-1],)

    def dot(self, axes, radius=0.08):
        return Dot3D(axes.c2p(*self.coords), radius=radius, color=self.color)

    def x_line(self, axes):
        return axes.get_vertical_line(axes.c2p(*self.projXY), color=self.color)
    
    def y_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.projXY), color=self.color)
    
    def z_line(self, axes):
        return Line3D(start=axes.c2p(*self.projZ), end=axes.c2p(*self.coords), thickness=0.01)
    
    def v_line(self, axes):
        return Line3D(start=axes.c2p(*self.projXY), end=axes.c2p(*self.coords), thickness=0.01)
    
    def xy_line(self, axes):
        return Line3D(start=np.array([0, 0, 0]), end=axes.c2p(*self.projXY), thickness=0.01)
    
    def proj_x_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.projXY), color=self.color)
    
    def proj_y_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.coords), color=self.color)

    def proj_z_line(self, axes):
        return Line3D(start=axes.c2p(*self.projZ), end=axes.c2p(*self.coords), thickness=0.01)

    def arrow(self, axes):
        return Arrow3D(
            start=np.array(axes.c2p(0, 0, 0)),
            end=np.array(axes.c2p(*self.coords)),
            resolution=8,
            color=self.color)

@dataclass
class VideoStatus:
    time: float = 0
    videoObject: cv2.VideoCapture = None
    def __deepcopy__(self, memo):
        return self

class VideoMobject(ImageMobject):
    '''
    Following a discussion on Discord about animated GIF images.
    Modified for videos

    Parameters
    ----------
    filename
        the filename of the video file

    imageops
        (optional) possibility to include a PIL.ImageOps operation, e.g.
        PIL.ImageOps.mirror

    speed
        (optional) speed-up/slow-down the playback

    loop
        (optional) replay the video from the start in an endless loop

    https://discord.com/channels/581738731934056449/1126245755607339250/1126245755607339250
    2023-07-06 Uwe Zimmermann & Abulafia
    2024-03-09 Uwe Zimmermann
    '''
    def __init__(self, filename=None, imageops=None, speed=1.0, loop=False, **kwargs):
        self.filename = filename
        self.imageops = imageops
        self.speed = speed
        self.loop = loop
        self._id = id(self)
        self.status = VideoStatus()
        self.status.videoObject = cv2.VideoCapture(filename)

        self.status.videoObject.set(cv2.CAP_PROP_POS_FRAMES, 1)
        ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)            
            img = Image.fromarray(frame)

            if imageops is not None:
                img = imageops(img)
        else:
            img = Image.fromarray(np.uint8([[63, 0, 0, 0],
                                        [0, 127, 0, 0],
                                        [0, 0, 191, 0],
                                        [0, 0, 0, 255]
                                        ]))
        super().__init__(img, **kwargs)
        if ret:
            self.add_updater(self.videoUpdater)

    def videoUpdater(self, mobj, dt):
        if dt == 0:
            return
        status = self.status
        status.time += 1000 * dt * mobj.speed
        self.status.videoObject.set(cv2.CAP_PROP_POS_MSEC, status.time)
        ret, frame = self.status.videoObject.read()
        if (ret == False) and self.loop:
            status.time = 0
            self.status.videoObject.set(cv2.CAP_PROP_POS_MSEC, status.time)
            ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)  # needed here?
