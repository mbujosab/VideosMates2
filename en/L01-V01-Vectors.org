class L01_V01_E01_VectorsDefinition(VoiceoverScene):
    def construct(self):
        self.set_speech_service(AzureService(voice="en-GB-RyanNeural"))
        #self.set_speech_service( AzureService(voice="es-ES-AlvaroNeural" ) )
      
        #self.set_speech_service(GTTSService(lang="en", tld="com"))        
        myTemplate = TexTemplate()
        myTemplate.add_to_preamble(r"\usepackage{nacal}")

        # Lateral copyright
        <<copyrightLateral>>

        # Cover
        title = Title(r"Definition of a Vector in \R[n] and Notation",
                     tex_template = myTemplate,
                     font_size=70).set_color(BLUE)
        self.play(Write(title))
        self.wait(1.5)
        self.play(FadeOut(title))

        # Vector Definition
        definition = Tex("A vector in ",
                         r"$\R[n]$",
                         r" is an \emph{ordered list} of $n$ numbers",
                         tex_template = myTemplate,
                         ).to_edge(UP).set_width(13)
        with self.voiceover(text=r"A vector in R n is an ordered list of numbers.") as tracker:
            self.add(definition)

        # Clarification of notation
        with self.voiceover(text="The R indicates that the numbers are real.") as tracker:
            self.play(Circumscribe(definition[1][0], fade_out=True), run_time=tracker.duration)

        with self.voiceover(text="""And the superscript n indicates that the list contains n
        numbers.""") as tracker:
            self.play(Circumscribe(definition[1][1], fade_out=True), run_time=tracker.duration)
            self.wait(0.3)

        with self.voiceover(text="""that the list being ordered means that the order in which the 
        elements appear matters.""") as tracker:
            self.play(Circumscribe(definition[2][4:15], fade_out=True), run_time=tracker.duration)

        # Examples
        Ej = Tex(r"\textbf{Examples:}",
                 tex_template = myTemplate,
                 font_size=50).set_color(GREEN).next_to(definition, DOWN, aligned_edge=LEFT)
        self.add(Ej)

        d = nc.Vector([1,2,3],'h')
        Ej1 = MathTex( d.latex(), "\\ne", nc.Vector(reversed(d),'h').latex() ) 
        with self.voiceover(text="""For example, the vectors, one two three and three two one, are
        different.""") as tracker:
            self.play(FadeIn(Ej1))
            self.wait(tracker.duration-0.5)
            self.play(FadeOut(Ej1))

        p = nc.Vector([sp.pi,sp.pi,sp.pi,sp.pi],'h')
        Ej2 = MathTex(p.latex(), r"\ne", nc.Vector(p|(1,2),'h').latex() )
        with self.voiceover(text=r"""Also, vectors with a different number of
        elements are different. For instance, the vector on the left belongs to R 4 as it is a list of
        4 numbers. The one on the right belongs to R
        2.""") as tracker:
            self.add(Ej2)
            self.wait(tracker.duration-0.5)
            self.play(FadeOut(Ej2))

        c = nc.Vector([sp.pi,nc.fracc(3,4),0,0.11],'h')
        Ej3 = MathTex(c.latex(),"=",c.latex())
        with self.voiceover(text=r"""Consequently, two vectors will be equal if, and only if, their
        respective lists are identical""") as tracker:
            self.play(FadeIn(Ej3))
            self.wait(tracker.duration)
            self.play(FadeOut(Ej3))            
        self.play(FadeOut(Ej))
        self.wait()

class L01_V01_E02_VectorsNotation(VoiceoverScene):
    def construct(self):
        self.set_speech_service( AzureService(voice="en-GB-RyanNeural" ) )
      
        #self.set_speech_service(GTTSService(lang="es", tld="com"))        
        myTemplate = TexTemplate()
        myTemplate.add_to_preamble(r"\usepackage{nacal}")

        # Lateral copyright
        <<copyrightLateral>>

        # Vector Definition
        definition = Tex("A vector in ",
                         r"$\R[n]$",
                         r" is an \emph{ordered list} of $n$ numbers",
                         tex_template = myTemplate,
                         ).to_edge(UP).set_width(13)
        self.add(definition)
        d = nc.Vector([1,2,3],'h')
        p = nc.Vector([sp.pi,sp.pi,sp.pi,sp.pi],'h')

        # Notation
        Notac = Tex(r"\textbf{Notation:}",
                 tex_template = myTemplate,
                 font_size=50).set_color(BLUE).next_to(definition, DOWN, aligned_edge=LEFT)
        self.add(Notac)
        self.wait()

        a    = nc.Vector( [5, 1, 10] , 'h')        
        Not1 = MathTex(a.latex(),        tex_template = myTemplate,)
        Not2 = MathTex(r"=",             tex_template = myTemplate,)
        Not3 = MathTex(a.copy().latex(), tex_template = myTemplate,)
        grp1 = VGroup(Not1,Not2,Not3).arrange(RIGHT)
        with self.voiceover(text=r"""To express a vector, it is enough to indicate the list of elements
        in their corresponding order.""") as tracker:
            self.add(grp1)
            self.play(Circumscribe(definition[2][4:15]),run_time=tracker.duration)

        with self.voiceover(text=r"""For this reason, we can write the same vector both horizontally and
        vertically. But keep in mind that most books do not respect this convention and consider,
        contrary to what we will do, that row vectors and column vectors are distinct objects.""") as tracker:
            self.wait(tracker.duration+0.3)

        with self.voiceover(text=r"""We will always write the list of numbers enclosed in parentheses,
        placing a comma after each element when writing the vector horizontally.""") as tracker:
            self.play(Indicate(grp1[0][0][::len(grp1[0][0])-1]),
                      Indicate(grp1[2][0][0:2]), Indicate(grp1[2][0][-2:]),
                      run_time=tracker.duration/2)
            self.play(Flash(grp1[0][0][2]),
                      Flash(grp1[0][0][4]), Flash(grp1[0][0][7]),
                      run_time=tracker.duration/8)
            self.wait(tracker.duration/8)
            self.play(Circumscribe(grp1[0]))
            self.play(FadeOut(grp1))

        VectorNoNumero =  MathTex(r"(3)",
                                  r"\ne",(3*nc.V1(1)).latex(),
                                  r"\in\R[1]",
                                  tex_template = myTemplate,)
        with self.voiceover(text=r"""Thus, we can distinguish a number in parentheses from a
        vector in R 1;""") as tracker:
            self.add(VectorNoNumero)
            self.play(Indicate(VectorNoNumero[0]),run_time=tracker.duration*2/3)
            self.play(Indicate(VectorNoNumero[2]),run_time=tracker.duration/3)
        with self.voiceover(text=r"""That is, from a list containing a single number.""") as tracker:
            self.play(Indicate(VectorNoNumero[3]),
                      Flash(definition[2][-8]),
                      run_time=tracker.duration)
            self.play(FadeOut(VectorNoNumero))

        Vectors = MathTex(r"\Vect{a}, \Vect{b}, \Vect{c},\ldots\Vect{x}, \Vect{y}, \Vect{z}",
                           tex_template = myTemplate,).move_to( UP )
        Vector1  = MathTex(r"\Vect{a}=",a.copy().latex(),   tex_template = myTemplate,)
        Vector2  = MathTex(r"\Vect{d}=",d.copy().latex(),   tex_template = myTemplate,)
        Vector3  = MathTex(r"\Vect{x}=",p.copy().latex(),   tex_template = myTemplate,)
        grp3 = VGroup(Vector1,Vector2,Vector3).arrange(RIGHT, buff=2).next_to(Vectors, DOWN)
        with self.voiceover(text=r"""To denote vectors, we will use lowercase letters in bold
        italics.""") as tracker:
            self.add(Vectors)
            self.add(grp3)
            self.wait(tracker.duration/2)
            self.play(Indicate(Vectors),run_time=tracker.duration/2)
            self.play(FadeOut(Vectors))
            self.play(Indicate(Vector1[0][0],scale_factor=2.),
                      Indicate(Vector2[0][0],scale_factor=2.),
                      Indicate(Vector3[0][0],scale_factor=2.),
                      run_time=1.5)
            self.play(FadeOut(grp3))

        Vzero = MathTex(r"\Vect{0}", tex_template = myTemplate,)#.move_to( UP )
        with self.voiceover(text=r"""A bold zero denotes a vector whose components are all
        zeros.""") as tracker:
            self.add(Vzero)
            self.play(Indicate(Vzero))
            self.wait(tracker.duration/2)
            self.play(FadeOut(Vzero))

        Vzero1 = MathTex(r"\Vect{0}=", nc.V0(1).latex(), ",",  tex_template = myTemplate,)
        Vzero2 = MathTex(r"\Vect{0}=", nc.V0(2).latex(), ",",  tex_template = myTemplate,)
        Vzero3 = MathTex(r"\Vect{0}=", nc.V0(3).latex(), ",",  tex_template = myTemplate,)
        Vzero6 = MathTex(r"\Vect{0}=", nc.V0(6).latex(), ",",  tex_template = myTemplate,)
        VzeroN = MathTex(r"\Vect{0}\in\R[100]",             tex_template = myTemplate,)
        grp2   = VGroup(Vzero1,Vzero2,Vzero3,Vzero6,VzeroN).arrange(RIGHT, buff=0.7)
        with self.voiceover( text = r"""Note that a bold zero does not indicate its number of
        components. Usually, the number of zeros is inferred from context.""" ) as tracker:
            self.add(grp2)
            self.wait(tracker.duration)
            self.play(FadeOut(grp2),FadeOut(Notac),FadeOut(definition))
            self.wait(1.5)

def credits(self, variant=1):
    def analyze_chosen_option(type):
        'Analysis of the chosen elimination options'
        list_options = [100, 20, 10, 4, 2, 1]
        option_set = set()
        for t in list_options:
            if (type - (type % t)) in list_options:
                option_set.add(type - (type % t))
                type = type % t
        return option_set    
    copyright = Tex(r"\textcopyright{\;} 2026\; Marcos Bujosa")
    if 1 in analyze_chosen_option(variant):
        stampRight  = VGroup(copyright.copy()).rotate(PI/2).scale(0.5).to_edge(RIGHT, buff=0.1).set_color(GRAY_D)
        self.add(stampRight)
    if 2 in analyze_chosen_option(variant):
        stampLeft  = VGroup(copyright.copy()).rotate(-PI/2).scale(0.5).to_edge(LEFT, buff=0.1).set_color(GRAY_D)
        self.add(stampLeft)
    if 4 in analyze_chosen_option(variant):

class VectorR2():
    def __init__(self, list, rpr='c', color=GRAY_B):
        """Initializes a Vector with a list"""
        coords = list + [0] if len(list) < 3 else list        
        self.color = color
        #self.coords = tuple(coords)
        #self.coords = tuple([float(i) for i in coords])
        self.coords = tuple([int(i) if float(i) == int(i) else float(i) for i in coords])
        self.Vector = nc.Vector(self.coords[:2], rpr)
        self.tex = MathTex(self.Vector.latex(), color=self.color).scale(0.8)

    def dot(self, axes, radius=0.08):
        return Dot(axes.c2p(*self.coords), radius=radius, color=self.color)

    def v_line(self, axes):
        return axes.get_vertical_line(axes.c2p(*self.coords), color=self.color)

    def h_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.coords), color=self.color)

    def arrow(self, axes):
        return axes.get_vector(self.Vector.lista, stroke_color=self.color, stroke_width=4)

#+name: Definition of VectorR3
#+BEGIN_SRC python :results discard

class VectorR3():
    def __init__(self, list, rpr='c', color=GRAY_B):
        """Initializes a Vector with a list"""
        coords = list + [0] if len(list) < 3 else list
        self.color = color
        #self.coords = tuple(coords)
        self.coords = tuple([int(i) if float(i) == int(i) else float(i) for i in coords])
        self.Vector = nc.Vector(self.coords, rpr)
        self.tex = MathTex(self.Vector.latex(), color=self.color).scale(0.8)
        self.projXY = self.coords[:-1] + (0,)
        self.projZ  = (0, 0,) + (self.coords[-1],)

    def dot(self, axes, radius=0.08):
        return Dot3D(axes.c2p(*self.coords), radius=radius, color=self.color)

    def x_line(self, axes):
        return axes.get_vertical_line(axes.c2p(*self.projXY), color=self.color)
    
    def y_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.projXY), color=self.color)
    
    def z_line(self, axes):
        return Line3D(start=axes.c2p(*self.projZ), end=axes.c2p(*self.coords), thickness=0.01)
    
    def v_line(self, axes):
        return Line3D(start=axes.c2p(*self.projXY), end=axes.c2p(*self.coords), thickness=0.01)
    
    def xy_line(self, axes):
        return Line3D(start=np.array([0, 0, 0]), end=axes.c2p(*self.projXY), thickness=0.01)
    
    def proj_x_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.projXY), color=self.color)
    
    def proj_y_line(self, axes):
        return axes.get_horizontal_line(axes.c2p(*self.coords), color=self.color)

    def proj_z_line(self, axes):
        return Line3D(start=axes.c2p(*self.projZ), end=axes.c2p(*self.coords), thickness=0.01)

    def arrow(self, axes):
        return Arrow3D(
            start=np.array(axes.c2p(0, 0, 0)),
            end=np.array(axes.c2p(*self.coords)),
            resolution=8,
            color=self.color)

#+name: Definition of VideoStatus and VideoMobject
#+BEGIN_SRC python :results discard
import cv2 # needs opencv-python https://pypi.org/project/opencv-python/
from PIL import Image, ImageOps
from dataclasses import dataclass

@dataclass
class VideoStatus:
    time: float = 0
    videoObject: cv2.VideoCapture = None
    def __deepcopy__(self, memo):
        return self

class VideoMobject(ImageMobject):
    '''
    Following a discussion on Discord about animated GIF images.
    Modified for videos

    Parameters
    ----------
    filename
        the filename of the video file

    imageops
        (optional) possibility to include a PIL.ImageOps operation, e.g.
        PIL.ImageOps.mirror

    speed
        (optional) speed-up/slow-down the playback

    loop
        (optional) replay the video from the start in an endless loop

    https://discord.com/channels/581738731934056449/1126245755607339250/1126245755607339250
    2023-07-06 Uwe Zimmermann & Abulafia
    2024-03-09 Uwe Zimmermann
    '''
    def __init__(self, filename=None, imageops=None, speed=1.0, loop=False, **kwargs):
        self.filename = filename
        self.imageops = imageops
        self.speed = speed
        self.loop = loop
        self._id = id(self)
        self.status = VideoStatus()
        self.status.videoObject = cv2.VideoCapture(filename)

        self.status.videoObject.set(cv2.CAP_PROP_POS_FRAMES, 1)
        ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)            
            img = Image.fromarray(frame)

            if imageops is not None:
                img = imageops(img)
        else:
            img = Image.fromarray(np.uint8([[63, 0, 0, 0],
                                        [0, 127, 0, 0],
                                        [0, 0, 191, 0],
                                        [0, 0, 0, 255]
                                        ]))
        super().__init__(img, **kwargs)
        if ret:
            self.add_updater(self.videoUpdater)

    def videoUpdater(self, mobj, dt):
        if dt == 0:
            return
        status = self.status
        status.time += 1000 * dt * mobj.speed
        self.status.videoObject.set(cv2.CAP_PROP_POS_MSEC, status.time)
        ret, frame = self.status.videoObject.read()
        if (ret == False) and self.loop:
            status.time = 0
            self.status.videoObject.set(cv2.CAP_PROP_POS_MSEC, status.time)
            ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)  # needed here?        
            img = Image.fromarray(frame)

            if mobj.imageops is not None:
                img = mobj.imageops(img)
            mobj.pixel_array = change_to_rgba_array(
                np.asarray(img), mobj.pixel_array_dtype
            )
