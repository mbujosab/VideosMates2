class L01_V02_E01_SumOfVectors(MySceneWithVoice):    
    def construct(self):
        self.set_speech_service(AzureService(voice="en-GB-RyanNeural"))
        #self.set_speech_service(AzureService(voice="en-US-JennyNeural"))       
        #self.set_speech_service(GTTSService(lang="en", tld="com"))       
        myTemplate = TexTemplate()
        myTemplate.add_to_preamble(r"""\usepackage{nacal} """)

        def slow(text, rate="-10%"):
            return f'<speak><prosody rate="{rate}">{text}</prosody></speak>'

        self.credits()
        
        # Cover page
        title = Title(_("Sum of Vectors in ") + r"\R[n]",
                     tex_template=myTemplate,
                     font_size=70).set_color(BLUE)
        self.play(Write(title))
        self.medium_pause()
        self.play(FadeOut(title))
	
        # Definition of vector sum
        operationSum = Tex(_("Sum of Vectors"),
                         tex_template=myTemplate, font_size=70
                         ).to_edge(UP).set_color(BLUE)

        operationDescription = Tex(_("The sum is defined component-wise."),
                         tex_template=myTemplate,
                         ).move_to([0, 2.5, 0]).to_edge(LEFT)
        # Examples
        ExR3 = Tex(r"\textbf{" + _("Example in ") + r"\R[3]:}",
                 tex_template=myTemplate,
                 font_size=50).set_color(GREEN).next_to(operationDescription, DOWN, aligned_edge=LEFT)

        ExR4 = Tex(r"\textbf{" + _("Example in ") + r"\R[4]:}",
                 tex_template=myTemplate,
                 font_size=50).set_color(GREEN).next_to(operationDescription, DOWN, aligned_edge=LEFT)
        
        a = nc.Vector([0, 3, 6])        
        b = nc.Vector([5, 1, 2])        
        s1 = MathTex(a.latex(), tex_template=myTemplate,)
        plus = MathTex(r"+", tex_template=myTemplate,)
        s3 = MathTex(b.latex(), tex_template=myTemplate,)
        equal = MathTex(r"=", tex_template=myTemplate,)
        s5 = MathTex((a + b).latex(), tex_template=myTemplate,)
        grp1 = VGroup(s1, plus, s3, equal, s5).arrange(RIGHT)
       
        self.add(operationSum)
        self.add(operationDescription)
        
        with self.voiceover(text=_("""We can add two vectors only if they have the same number of components.""")) as tracker:
            self.add(ExR3)
            self.add(grp1[0])
            self.add(grp1[2])

        with self.voiceover(text=_("""The result is another vector, defined component-wise.""")) as tracker:
            self.add(grp1[1])
            self.medium_pause()
            self.add(grp1[3])
            self.add(grp1[4][0][:2])
            self.add(grp1[4][0][-2:])

        with self.voiceover(text=_("""The first element is the sum of the first components of both vectors.""")) as tracker:
            self.play(FadeIn(grp1[4][0][2]), run_time=tracker.duration/3)
            self.play(Circumscribe(grp1[0][0][2]), Circumscribe(grp1[2][0][2]), run_time=tracker.duration*2/3)
            self.short_pause()
            
        with self.voiceover(text=_("""The second is the sum of the second components.""")) as tracker:
            self.play(FadeIn(grp1[4][0][3]), run_time=tracker.duration/3)
            self.play(Circumscribe(grp1[0][0][3]), Circumscribe(grp1[2][0][3]), run_time=tracker.duration*2/3)
            self.short_pause()
            
        with self.voiceover(text=_("""This process continues similarly for all components of the resulting vector.""")) as tracker:            
            self.play(FadeIn(grp1[4][0][4]))
            self.short_pause()
            self.play(Circumscribe(grp1[0][0][4]), Circumscribe(grp1[2][0][4]))
            self.short_pause(.3)
            self.play(FadeOut(grp1))

        self.pause()

        generic_vector_a = nc.Vector(sp.symbols('a:5')[1:])
        vga = MathTex(generic_vector_a.latex(), tex_template=myTemplate)
        
        generic_vector_b = nc.Vector(sp.symbols('b:5')[1:])
        vgb = MathTex(generic_vector_b.latex(), tex_template=myTemplate)

        vgab = MathTex((generic_vector_a + generic_vector_b).latex(), tex_template=myTemplate)

        grp2 = VGroup(vga, plus, vgb, equal, vgab).arrange(RIGHT)
        with self.voiceover(text=_("""Therefore, we can express the sum of vectors with 4 components in this way:""")) as tracker:
            self.play(FadeTransform(ExR3, ExR4))
            self.play(FadeIn(grp2))
            self.pause()

        with self.voiceover(text=_("""To define the sum in R n, a different approach is needed; one that avoids listing all components explicitly. This is particularly important when N is large.""")) as tracker:
            self.wait(tracker.duration/3)
            self.play(Indicate(vga[0][4:-4]), Indicate(vgb[0][4:-4]), Indicate(vgab[0][4:-4]), run_time=2)
            self.short_pause()
            
        Defn = Tex(r"\textbf{" + _("Definition in ") + r"\R[n]:}",
                 tex_template=myTemplate,
                 font_size=50).set_color(RED).next_to(operationDescription, DOWN, aligned_edge=LEFT)

        with self.voiceover(text=_("""An alternative is to define the sum using the notation discussed in the previous video. This allows us to express the following:""")) as tracker:
            self.play(FadeOut(grp2), FadeOut(ExR4), run_time=tracker.duration/3)
            self.add(Defn)

        cvab = MathTex(r"\elemRp{\Vect{a}+\Vect{b}}{i}", tex_template=myTemplate)
        cva  = MathTex(r"\eleVR{a}{i}", tex_template=myTemplate)
        cvb  = MathTex(r"\eleVR{b}{i}", tex_template=myTemplate)
        sum_eq = VGroup(cvab, equal, cva, plus, cvb).arrange(RIGHT).scale(1.5)
        
        where = Tex(_("where"))
        indices = MathTex(r"i=1:n", tex_template=myTemplate)
        pc_indices = VGroup(where, indices).arrange(RIGHT, buff=1)
        grp3 = VGroup(sum_eq, pc_indices).arrange(RIGHT, buff=1)

        with self.voiceover(text=_("""The index-i component of the sum vector""")) as tracker:
            self.play(FadeIn(grp3[0][:1],
                             scale=1.5,
                             rate_func=rate_functions.exponential_decay),
                      run_time=2*tracker.duration/5)
            self.pause(3*tracker.duration/5)
            
        with self.voiceover(text=_("""is equal to the sum of the index-i components of the original vectors.""")) as tracker:
            self.play(FadeIn(grp3[0][1:],
                             scale=0.5,
                             rate_func=rate_functions.exponential_decay),
                      run_time=2*tracker.duration/5) 
            self.pause(3*tracker.duration/5)
           
        with self.voiceover(text=_(r"""(where the index traverses the natural numbers from 1 to n).""")) as tracker:
            self.play(FadeIn(grp3[1]))
            self.short_pause()

        with self.voiceover(text=_(r"""This abstract definition is useful for demonstrating properties of vector operations, establishing our first rule for symbolic manipulation:""")) as tracker:
            self.pause(tracker.duration*2/3)
            self.play(Indicate(sum_eq[0][0][0]),
                      Indicate(sum_eq[0][0][-3:]),
                      Indicate(sum_eq[2][0][-2:]),
                      Indicate(sum_eq[4][0][-2:]),
                      run_time=tracker.duration/3)
            self.very_short_pause()


        with self.voiceover(text=_(r"""The sum of the index-i components""")) as tracker:            
            source0 = MathTex(r"\eleVR{a}{i}+\eleVR{b}{i}",
                              tex_template=myTemplate).next_to(grp3, DOWN, buff=1.2).scale(2)[0]
            target0 = MathTex(r"\elemRp{\Vect{a}+\Vect{b}}{i}",
                              tex_template=myTemplate).next_to(grp3, DOWN, buff=1.2).scale(2)[0]
            source1 = target0.copy()
            target1 = source0.copy()
            
            VGroup(source0, target0)
            self.add(source0)
            self.very_short_pause()
            
        with self.voiceover(text=_(r"""can be substituted with the index-i component of the resulting sum vector.""")) as tracker:

                                [1, 0, 4, 2, 3, 5, 6]]

self.credits(7)
	
operationSum = Tex(_("Sum of Vectors in ") + r"\R[n]",
                    tex_template=myTemplate,
                    font_size=70).to_edge(UP).set_color(BLUE)        
self.add(operationSum)

source0 = MathTex(r"\elemRp{\Vect{a}+\Vect{b}}{i}", tex_template=myTemplate).scale(2)[0]
source1 = source0.copy()
target0 = MathTex(r"\elemRp{\Vect{b}+\Vect{a}}{i}", tex_template=myTemplate).scale(2)[0]
equal = MathTex(r"=", tex_template=myTemplate,).scale(2)[0]
vgr0 = VGroup(source0, equal, source1).arrange(RIGHT, buff=1)
vgr1 = VGroup(source0, equal, target0).arrange(RIGHT, buff=1)
transform_index0 = [[1, 2, 3],
                    [3, 2, 1]]

with self.voiceover(text=_("""Now, let's use this rule to show the commutative property. As you'll see, the order in which we add the vectors does not change the final result.""")) as tracker:
    self.play(FadeIn(source0[1:-3]), FadeIn(vgr1[1]), FadeIn(source1[1:-3]))
    self.pause(tracker.duration/2)
    self.play(
        ,*[
            ReplacementTransform(vgr0[2][i], vgr1[2][j], rate_func=rate_functions.smooth)
            for i, j in zip(*transform_index0)
        ],
        run_time=tracker.duration/2)
    self.short_pause()
    
with self.voiceover(text=_("""We already know that two vectors are equivalent if their corresponding components are equal; hence, to confirm this equality, we must establish it component-wise.""")) as tracker:           
    self.pause(tracker.duration*2/5)
    self.play(FadeIn(source0[0]), FadeIn(source0[-3:]), FadeIn(target0[0]), FadeIn(target0[-3:]))
    self.pause(tracker.duration/4)
    self.play(Indicate(source0[-2:]), Indicate(target0[-2:]), run_time=tracker.duration/4)
    self.short_pause()

with self.voiceover(text=_("""Let’s begin by writing down the left-hand side. We will then manipulate it""")) as tracker:            
    self.play(Indicate(vgr0[0]),
              run_time=2*tracker.duration/3)

with self.voiceover(text=_("""until we reach the expression on the opposite side.""")) as tracker:            
    self.play(Indicate(vgr1[2]))

                type = type % t

#+name: Definition of VectorR2
#+BEGIN_SRC python :results discard

class VectorR2():
    def __init__(self, list, rpr='c', color=GRAY_B):
        """Initializes a Vector with a list"""

        self.projZ  = (0, 0,) + (self.coords[-1],)

    ----------
    filename
        the filename of the video file

    imageops
        (optional) possibility to include a PIL.ImageOps operation, e.g.
        PIL.ImageOps.mirror

    speed
        (optional) speed-up/slow-down the playback

    loop

    2023-07-06 Uwe Zimmermann & Abulafia

        self.filename = filename
        self.imageops = imageops
        self.speed = speed
        self.loop = loop
        self._id = id(self)
        self.status = VideoStatus()
        self.status.videoObject = cv2.VideoCapture(filename)

        self.status.videoObject.set(cv2.CAP_PROP_POS_FRAMES, 1)
        ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)            
            img = Image.fromarray(frame)

            if imageops is not None:
                img = imageops(img)
        else:
            img = Image.fromarray(np.uint8([[63, 0, 0, 0],
                                        [0, 127, 0, 0],
                                        [0, 0, 191, 0],
                                        [0, 0, 0, 255]
                                        ]))

            self.add_updater(self.videoUpdater)

    def videoUpdater(self, mobj, dt):

            ret, frame = self.status.videoObject.read()
        if ret:
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)  # needed here?

    colors = itertools.cycle([RED, TEAL, GREEN, BLUE, PURPLE])
    for i in range(len(tex)):
        n = Text(f"{i}", color=next(colors)).scale(0.7)
        n.next_to(tex[i], DOWN, buff=0.01)
        ni.add(n)
    return ni
