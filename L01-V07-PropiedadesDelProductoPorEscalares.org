PROPIEDADES_REALES = [r"a+b=b+a", r"a+(b+c)=(a+b)+c", r"a+0=a", r"a+(-a)=0",
                      r"ab=ba", r"a(b+c)=ab+ac", r"a(bc)=(ab)c", r"1a=a"]

PROPIEDADES_SUMA = [r"\Vect{a}+\Vect{b}=\Vect{b}+\Vect{a}",
                    r"\Vect{a}+(\Vect{b}+\Vect{c})=(\Vect{a}+\Vect{b})+\Vect{c}",
                    r"\Vect{a}+\Vect{0}=\Vect{a}",
                    r"\Vect{a}+(-\Vect{a})=\Vect{0}"]

PROPIEDADES_PRODUCTO = [r"\lambda(\Vect{a}+\Vect{b})=\lambda\Vect{a}+\lambda\Vect{b}",
                        r"(\lambda+\varphi)\Vect{a}=\lambda\Vect{a}+\varphi\Vect{a}",
                        r"\lambda(\varphi\Vect{a})=(\lambda\varphi)\Vect{a}",
                        r"1\Vect{a}=\Vect{a}"]

CAMPANITA = os.path.join(os.path.dirname(os.path.abspath(__file__)), "sonidos", "campanita.wav")   # suena al subir un contador

DEFINICIONES = [("suma",     r"\elemRp{\Vect{a}+\Vect{b}}{i}=\eleVR{a}{i}+\eleVR{b}{i}"),
                ("producto", r"\elemRp{\lambda\Vect{a}}{i}=\lambda\eleVRpE{a}{i}")]


class Panel:
    """Panel fijo de herramientas: definiciones a la izquierda y propiedades de los reales a la
    derecha. Las piezas se identifican por clave ("suma", "producto", 5, 6, ...)."""
    def __init__(self, escena, reales=range(5, 9), atenuadas=(), y_sup=2.95, font_size=28):
        self.escena, self.piezas, self.usos, self.contadores = escena, {}, {}, {}
        izq = VGroup(*[MathTex(f, tex_template=myTemplate, font_size=font_size) for _, f in DEFINICIONES])
        izq.arrange(DOWN, aligned_edge=LEFT, buff=0.22)
        for (clave, _), m in zip(DEFINICIONES, izq):
            self.piezas[clave] = m
        der = VGroup(*[MathTex(f"{k}.\\;", PROPIEDADES_REALES[k-1], tex_template=myTemplate, font_size=font_size)
                       for k in reales]).arrange(DOWN, aligned_edge=LEFT, buff=0.22)
        for k, m in zip(reales, der):
            self.piezas[k] = m
        cab_izq = Tex(tr("Definiciones", "Definitions"), font_size=24).set_color(GRAY_B)
        cab_der = Tex(tr("Números reales", "Real numbers"), font_size=24).set_color(GRAY_B)
        self.izq = VGroup(cab_izq, izq).arrange(DOWN, aligned_edge=LEFT, buff=0.2).to_edge(LEFT, buff=0.35)
        self.der = VGroup(cab_der, der).arrange(DOWN, aligned_edge=LEFT, buff=0.2).to_edge(RIGHT, buff=1.1)
        for g in (self.izq, self.der):
            g.align_to([0, y_sup, 0], UP)
        self.atenuadas = set(atenuadas)
        for clave in atenuadas:
            self.piezas[clave].set_opacity(0.3)
        self.todo = VGroup(self.izq, self.der)

    def enciende(self, *claves, run_time=0.8):
        """Trae al frente piezas atenuadas (cuando entran en juego)."""
        self.atenuadas -= set(claves)
        self.escena.play(*[self.piezas[c].animate.set_opacity(1) for c in claves], run_time=run_time)

    def pista(self, *claves, run_time=1):
        """La pista: las piezas que hacen falta se iluminan en amarillo y se recuadran."""
        self.atenuadas -= set(claves)
        self.marcos = VGroup(*[SurroundingRectangle(self.piezas[c], color=COLOR_ESCALAR, buff=0.1,
                                                    corner_radius=0.1, stroke_width=2) for c in claves])
        self.escena.play(*[self.piezas[c].animate.set_color(COLOR_ESCALAR).set_opacity(1) for c in claves],
                         Create(self.marcos), run_time=run_time)

    def senala(self, clave, run_time=0.8):
        """Antes de dar un paso: la pieza que se va a usar parpadea."""
        self.escena.play(Indicate(self.piezas[clave], color=COLOR_RESULTADO, scale_factor=1.12), run_time=run_time)

    def cuenta(self, clave, run_time=0.5):
        """Después del paso: sube el contador de usos de la pieza, con una campanita."""
        pieza = self.piezas[clave]
        self.usos[clave] = self.usos.get(clave, 0) + 1
        # insignia: disco de color con el número de usos dentro (no se confunde con parte de la fórmula)
        disco = Circle(radius=0.17, fill_color=COLOR_TITULO, fill_opacity=1, stroke_width=0)
        numero = Tex(f"\\textbf{{{self.usos[clave]}}}", font_size=24).set_color(WHITE).move_to(disco)
        nuevo = VGroup(disco, numero).next_to(pieza, RIGHT, buff=0.3).shift(0.1*UP)
        self.escena.add_sound(CAMPANITA)
        if clave in self.contadores:
            self.escena.play(ReplacementTransform(self.contadores[clave], nuevo), run_time=run_time)
        else:
            self.escena.play(FadeIn(nuevo, scale=0.5), run_time=run_time)
        self.contadores[clave] = nuevo


def cabecera_y_enunciado(escena, texto, numero):
    """Cabecera de la escena (lleva el número de la propiedad) y enunciado de la propiedad `numero`
    (5 a 8), sin numerar, en el hueco entre los dos lados del panel."""
    cabecera = Tex(texto, tex_template=myTemplate, font_size=36).set_color(COLOR_TITULO).to_edge(UP, buff=0.25)
    enunciado = MathTex(PROPIEDADES_PRODUCTO[numero-5], tex_template=myTemplate, font_size=40).set_color(COLOR_RESULTADO)
    enunciado.scale_to_fit_width(min(enunciado.width, 4.6)).move_to([-0.3, 2.2, 0])
    escena.add(cabecera)
    return cabecera, enunciado


def linea(*formula, de_reales=False):
    """Una línea de la cadena de igualdades (en teal si el paso es una operación entre números reales);
    en varios trozos si hay que resaltar una parte."""
    return MathTex(*formula, tex_template=myTemplate, font_size=36).set_color(COLOR_REGLA if de_reales else WHITE)


def coloca_cadena(cadena, y_sup=0.55, salto=0.66, x_izq=-3.3, sangria=0.6):
    """Coloca la cadena bajo el enunciado y entre los dos lados del panel: la primera línea en x_izq y
    las siguientes (que empiezan por '=') alineadas entre sí y desplazadas `sangria` a la derecha."""
    for k, l in enumerate(cadena):
        l.move_to([0, y_sup - k*salto, 0]).align_to([x_izq + (sangria if k else 0), 0, 0], LEFT)
    return cadena


def paso(escena, l, panel, clave, run_time=1, veces=1):
    """Un paso de la cadena: parpadea la pieza que se va a usar (la voz la acaba de nombrar), se
    escribe la línea y sube el contador con la campanita (dos veces si se aplica en dos sumandos)."""
    panel.senala(clave)
    escena.play(Write(l), run_time=run_time)
    for _ in range(veces):
        panel.cuenta(clave)


def concluye(escena, enunciado, texto_es, texto_en, cadena=None):
    """Cierre de cada demostración: i es cualquiera, luego los vectores son iguales (en la marca 'cc'
    se destacan los dos extremos de la cadena) y al final se enciende el enunciado."""
    with escena.voiceover(text=tr(texto_es, texto_en)) as tracker:
        # manim-voiceover sitúa al final del audio una marca en la posición 0 del texto (interpola
        # antes de la primera palabra): una marca al principio significa "ya", sin esperar
        bt = {m: (tracker.start_t if tracker.bookmark_distances[m] == 0 else t) for m, t in tracker.bookmark_times.items()}
        if cadena is not None and "cc" in bt:
            if bt["cc"] - escena.renderer.time > 0.01: escena.wait(bt["cc"] - escena.renderer.time)
            rt = min(3, max(0.8, bt["ok"] - bt["cc"])) if "ok" in bt else 1.5   # dura lo que la frase, hasta el recuadro
            escena.play(Indicate(cadena[0], color=COLOR_ESCALAR, scale_factor=1.15),
                        Indicate(cadena[-1], color=COLOR_ESCALAR, scale_factor=1.15), run_time=rt)
        if "ok" in bt:
            if bt["ok"] - escena.renderer.time > 0.01: escena.wait(bt["ok"] - escena.renderer.time)
        # lo demostrado: recuadro amarillo y parpadeo del enunciado
        marco = SurroundingRectangle(enunciado, color=COLOR_ESCALAR, buff=0.2, corner_radius=0.15, stroke_width=3)
        escena.play(Create(marco), Indicate(enunciado, color=COLOR_RESULTADO, scale_factor=1.15), run_time=1.5)
        restante = tracker.start_t + tracker.duration - escena.renderer.time
        escena.pausa(max(0.3, restante))

** Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la
clase del vídeo completo): corregir una escena no obliga a renderizarlo
todo ni a pagar otra vez la voz, y los límites entre escenas quedan
exactos. Cada segmento se recodifica igualando la duración del audio y
del vídeo (concatenar sin recodificar deja un parón en cada unión, pues
manim genera el audio unas décimas más largo que la imagen). La música
suena en la portada (volumen =volumen=) y se funde cuando empieza la
primera escena; vuelve al empezar la escena =cierre= (bajo la voz, con
=volumen_voz=), sube en la última escena (los créditos) y se funde a
negro con la imagen. La voz se normaliza con =loudnorm=. Los =.srt= de las escenas se desplazan y se
funden en un único =video.srt= junto al mp4 (para subirlo a YouTube).

=escenas= es la lista ordenada de clases; =cierre= la escena en la que
vuelve la música. El mp4 y el srt finales quedan en la raíz del
repositorio del idioma.

#+name: montaje
#+BEGIN_SRC bash :var fichero="LXX_VYY_FileName" :var escenas="LXX_VYY_E00_Portada LXX_VYY_E01_Introduccion LXX_VYY_E02_TemaEscena LXX_VYY_E03_TemaEscena LXX_VYY_E04_TemaEscena LXX_VYY_E05_Resumen ZCreditosFinales" :var cierre="LXX_VYY_E05_Resumen" :var video="LXX_VYY_NombreVideo" :var idioma="es" :var calidad="1080p60" :var musica="music/3Blue1Brown-Zeta-X3.mp3" :var volumen="0.25" :var volumen_voz="0.04" :results output
export LC_ALL=C   # decimales con punto para bc y printf
[ "$idioma" = "en" ] && raiz=../videoMath2 || raiz=.
dir=$raiz/lib/media/videos/$fichero/$calidad
tmp=$raiz/tmp/$video; mkdir -p $tmp; rm -f $tmp/seg_*.mp4

# 1. Un segmento por escena, con audio y vídeo de la misma duración exacta
i=0; lista=""; offsets=""; T=0; T0=""; Tc=""
for esc in $escenas; do
  seg=$(printf "%s/seg_%02d.mp4" $tmp $i)
  dur=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=duration -of default=nk=1:nw=1 $dir/$esc.mp4)
  fps=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=r_frame_rate -of default=nk=1:nw=1 $dir/$esc.mp4)
  if ffprobe -loglevel error -select_streams a -show_entries stream=codec_type -of csv=p=0 $dir/$esc.mp4 | grep -q audio; then
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -af "aresample=48000,apad" -ac 2 -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  else
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -f lavfi -i anullsrc=r=48000:cl=stereo -map 0:v -map 1:a -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  fi
  d=$(ffprobe -loglevel error -show_entries format=duration -of default=nk=1:nw=1 $seg)
  offsets="$offsets $T"; [ "$esc" = "$cierre" ] && Tc=$T
  printf "%2d  %-55s %8.3f s  (empieza en %8.3f)\n" $i $esc $d $T
  T=$(bc -l <<< "$T+$d"); [ -z "$T0" ] && T0=$d; dl=$d
  lista="$lista $seg"; i=$((i+1))
done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]loudnorm=I=-16:LRA=11:TP=-1.5[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];
  [$((n+2)):a]atrim=$dur_m2:$fin_m3,asetpts=PTS-STARTPTS,volume=$volumen,afade=t=in:st=0:d=1,adelay=delays=${ms_m3}:all=1[m3];
  [voz][m1][m2][m3]amix=inputs=4:duration=first:normalize=0[mix];
  [mix]afade=t=out:st=$tfade:d=6[a];
  [v0]fade=t=out:st=$tfade:d=6[v]" \
  -map "[v]" -map "[a]" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -c:a aac -b:a 192k $raiz/$video.mp4
echo "Vídeo: $raiz/$video.mp4"

# 3. Subtítulos: desplazar el srt de cada escena y fundirlos en uno
ESCENAS="$escenas" OFFSETS="$offsets" python3 - "$dir" "$raiz/$video.srt" <<'EOF'
import sys, os, re
dir, salida = sys.argv[1], sys.argv[2]
escenas = os.environ["ESCENAS"].split(); offsets = [float(x) for x in os.environ["OFFSETS"].split()]
def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000
def s2t(x):
    ms = round(x*1000); h, r = divmod(ms, 3600000); m, r = divmod(r, 60000); s, ms = divmod(r, 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"
k, out = 1, []
for esc, off in zip(escenas, offsets):
    ruta = f"{dir}/{esc}.srt"
    if not os.path.exists(ruta): continue
    for bloque in re.split(r"\n\s*\n", open(ruta, encoding="utf-8").read().strip()):
        lineas = bloque.strip().splitlines()
        if len(lineas) < 3 or "-->" not in lineas[1]: continue
        a, _, b = lineas[1].partition("-->")
        out.append(f"{k}\n{s2t(t2s(a)+off)} --> {s2t(t2s(b)+off)}\n" + "\n".join(lineas[2:])); k += 1
open(salida, "w", encoding="utf-8").write("\n\n".join(out) + "\n")
print(f"Subtítulos: {salida} ({k-1} entradas)")
EOF
#+END_SRC

** Extraer el guion (la narración) para revisarlo

Recorre el fichero de python y saca, escena por escena, los textos
~tr(...)~ del idioma pedido: en prosa la narración (los que están dentro
de ~self.voiceover~) y entre corchetes los textos que aparecen en
pantalla. Sirve para revisar el guion sin leer código y para la
descripción o los subtítulos del vídeo. Como los dos py solo difieren
en =LANG=, lee siempre el fichero español (=./lib/=).

#+name: guion
#+BEGIN_SRC python :python python3 :var fichero="LXX_VYY_FileName" :var idioma="es" :results output
import ast, textwrap

arbol = ast.parse(open(f"lib/{fichero}.py", encoding="utf-8").read())
idx   = 0 if idioma == "es" else 1

def texto(nodo):
    try:
        return ast.literal_eval(nodo)
    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
#+END_SRC
