class Linea:
    """Una línea de cadena: una sola MathTex partida en trozos por su número de glifos."""
    _n = {}

    def __init__(self, trozos, font_size=34):
        self.trozos = list(trozos)
        self.tex = MathTex("".join(self.trozos), tex_template=myTemplate, font_size=font_size)
        self.g = self.tex[0]
        self.ini, k = [], 0
        for t in self.trozos:
            if t not in Linea._n:
                Linea._n[t] = len(MathTex(t, tex_template=myTemplate)[0])
            self.ini.append(k); k += Linea._n[t]
        assert k == len(self.g), (self.trozos, k, len(self.g))
        for t, trozo in enumerate(self.trozos):
            for j, c in COLORES.get(trozo, {}).items():
                self.g[self(t, j)].set_color(c)

    def __call__(self, t, k=None):
        if k is None:
            return list(range(self.ini[t], self.ini[t] + Linea._n[self.trozos[t]]))
        assert 0 <= k < Linea._n[self.trozos[t]], (t, k)
        return self.ini[t] + k

    def trozo(self, t):
        return VGroup(*[self.g[i] for i in self(t)])


def igual_trozo(lo, to, ld, td):
    """Un trozo que pasa entero, glifo a glifo (los dos trozos tienen los mismos glifos)."""
    return list(zip(lo(to), ld(td)))


def nace_l(escena, lo, ld, fijos=(), moviles=(), aparecen=(), tramo=None, marca=None, extra=(), t_fijos=0.8, t_moviles=0.9):
    """Una línea (o un tramo de una fila) nace de la anterior (V13). Índices de glifos de lo y ld."""
    t0, t1 = tramo or (0, len(ld.trozos) - 1)
    destino = [i for t in range(t0, t1 + 1) for i in ld(t)]
    igual = ld(t0)   # el tramo empieza por su igual
    cubiertos = sorted({j for _, j in fijos} | {j for _, j in moviles} | set(aparecen) | set(igual))
    assert cubiertos == destino, (sorted(set(destino) - set(cubiertos)), sorted(set(cubiertos) - set(destino)))
    escena.play(*extra, *[FadeIn(ld.g[j]) for j in igual],
                *[TransformFromCopy(lo.g[i], ld.g[j]) for i, j in fijos], run_time=t_fijos)
    if marca is not None:
        escena.wait_until_bookmark(marca)
    if moviles or aparecen:
        # si dos glifos de origen se reúnen en uno de destino, el segundo viaja como copia y se retira
        vistos, anims, sobran = {j for _, j in fijos}, [], []
        for i, j in moviles:
            if j in vistos:
                viajero = lo.g[i].copy()
                sobran.append(viajero)
                anims.append(Transform(viajero, ld.g[j].copy()))
            else:
                vistos.add(j)
                anims.append(TransformFromCopy(lo.g[i], ld.g[j]))
        escena.play(*anims, *[FadeIn(ld.g[j]) for j in aparecen], run_time=t_moviles)
        escena.remove(*sobran)


def entera(escena, linea):
    """Tras nacer glifo a glifo, la línea vuelve a estar en la escena como un solo objeto."""
    escena.remove(*linea.g); escena.add(linea.tex)


def coloca_cadena(lineas, y_sup=0.15, salto=0.56, sangria=0.6, ancho_max=12.6):
    """La primera línea arriba y las siguientes (que empiezan por '=') sangradas; todas a la misma escala y el
    conjunto centrado. Si la línea más ancha no cabe, todas se reducen igual."""
    ancho = max(l.tex.width for l in lineas[1:]) + sangria
    ancho = max(ancho, lineas[0].tex.width)
    if ancho > ancho_max:
        for l in lineas:
            l.tex.scale(ancho_max/ancho)
        sangria *= ancho_max/ancho; ancho = ancho_max
    x_izq = -ancho/2
    for k, l in enumerate(lineas):
        l.tex.move_to([0, y_sup - k*salto, 0]).align_to([x_izq + (sangria if k else 0), 0, 0], LEFT)

class Panel:
    """Panel de herramientas (V06, V07, L02-V01): las definiciones que pide la demostración a la izquierda y las
    propiedades de los números reales a la derecha, en pequeño. Cada uso se cuenta con una insignia y la campanita."""
    def __init__(self, escena, definiciones, reales, y_sup=2.95, font_size=28):
        self.escena, self.piezas, self.usos, self.contadores, self.cajas = escena, {}, {}, {}, {}
        izq = VGroup(*[MathTex(DEFINICIONES[c], tex_template=myTemplate, font_size=font_size) for c in definiciones])
        izq.arrange(DOWN, aligned_edge=LEFT, buff=0.22)
        for c, m in zip(definiciones, izq):
            self.piezas[c] = m
        der = VGroup(*[MathTex(f"{k}.\\;", REALES[k], tex_template=myTemplate, font_size=font_size) for k in reales])
        der.arrange(DOWN, aligned_edge=LEFT, buff=0.22)
        for k, m in zip(reales, der):
            self.piezas[k] = m
        cab_izq = Tex(tr("Definiciones", "Definitions"), font_size=24).set_color(GRAY_B)
        cab_der = Tex(tr("Números reales", "Real numbers"), font_size=24).set_color(GRAY_B)
        self.izq = VGroup(cab_izq, izq).arrange(DOWN, aligned_edge=LEFT, buff=0.2).to_edge(LEFT, buff=0.35)
        self.der = VGroup(cab_der, der).arrange(DOWN, aligned_edge=LEFT, buff=0.2).to_edge(RIGHT, buff=1.0)
        for g in (self.izq, self.der):
            g.align_to([0, y_sup, 0], UP)
        self.todo = VGroup(self.izq, self.der)

    def enciende(self, clave, run_time=0.4):
        """Al nombrarla al presentar el panel: la pieza crece un instante."""
        return Indicate(self.piezas[clave], color=COLOR_ESCALAR, scale_factor=1.12, run_time=run_time)

    def senala(self, clave):
        return Indicate(self.piezas[clave], color=COLOR_RESULTADO, scale_factor=1.12)

    def pista(self, *claves):
        """Las piezas que hacen falta pasan a amarillo y se recuadran (el amarillo solo no se ve en baja resolución)."""
        for c in claves:
            self.cajas[c] = marco(self.piezas[c])
        self.escena.play(*[self.piezas[c].animate.set_color(COLOR_ESCALAR) for c in claves],
                         *[Create(self.cajas[c]) for c in claves], run_time=0.8)

    def quita_pista(self):
        self.escena.play(*[self.piezas[c].animate.set_color(WHITE) for c in self.cajas],
                         *[FadeOut(b) for b in self.cajas.values()], run_time=0.5)
        self.cajas = {}

    def cuenta(self, clave, run_time=0.5):
        """Después del paso: sube el contador de usos de la pieza, con la campanita (insignia de V06)."""
        pieza = self.piezas[clave]
        self.usos[clave] = self.usos.get(clave, 0) + 1
        disco = Circle(radius=0.17, fill_color=COLOR_TITULO, fill_opacity=1, stroke_width=0)
        numero = Tex(f"\\textbf{{{self.usos[clave]}}}", font_size=24).set_color(WHITE).move_to(disco)
        nuevo = VGroup(disco, numero).move_to([pieza.get_right()[0] + 0.3 + 0.17, pieza.get_y() + 0.1, 0])
        self.escena.add_sound(CAMPANITA)
        if clave in self.contadores:
            self.escena.play(ReplacementTransform(self.contadores[clave], nuevo), run_time=run_time)
        else:
            self.escena.play(FadeIn(nuevo, scale=0.5), run_time=run_time)
        self.contadores[clave] = nuevo


def paso(escena, panel, claves, lo, ld, fijos=(), moviles=(), aparecen=(), marca=None, t_moviles=0.9):
    """Un paso de una demostración con panel (orden de V06): la pieza parpadea mientras se copia lo que no cambia,

           t_moviles=t_moviles)
    entera(escena, ld)
    for c in claves:
        panel.cuenta(c)

*** Escena 1 - La compra de Omega, dos días

El ejemplo que propuso Marcos (2026-11-21): con los precios fijos, el coste de la compra de las dos cestas juntas es
la suma de los costes, y el de la cesta doble, el doble. Las cestas y los precios se construyen en pantalla (el vídeo
se sostiene solo); la voz recuerda en una frase qué es el producto punto. Cada coste nace al decirlo; la cantinela de

** Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la
clase del vídeo completo): corregir una escena no obliga a renderizarlo
todo ni a pagar otra vez la voz, y los límites entre escenas quedan
exactos. Cada segmento se recodifica igualando la duración del audio y
del vídeo (concatenar sin recodificar deja un parón en cada unión, pues
manim genera el audio unas décimas más largo que la imagen). La música
suena en la portada (volumen =volumen=) y se funde cuando empieza la
primera escena; vuelve al empezar la escena =cierre= (bajo la voz, con
=volumen_voz=), sube en la última escena (los créditos) y se funde a
negro con la imagen. La voz se mide entera y se lleva a -16 LUFS con
una ganancia fija y un limitador (no con =loudnorm= en una pasada, que
subía el volumen poco a poco al empezar). Los =.srt= de las escenas se desplazan y se
funden en un único =video.srt= junto al mp4 (para subirlo a YouTube).

=escenas= es la lista ordenada de clases; =cierre= la escena en la que
vuelve la música (el resumen). El mp4 y el srt finales quedan en la raíz del repositorio del
idioma.

#+name: montaje
#+BEGIN_SRC bash :var fichero="LXX_VYY_FileName" :var escenas="LXX_VYY_E00_Portada LXX_VYY_E01_Introduccion LXX_VYY_E02_TemaEscena LXX_VYY_E03_TemaEscena LXX_VYY_E04_TemaEscena LXX_VYY_E05_Resumen ZCreditosFinales" :var cierre="LXX_VYY_E05_Resumen" :var video="LXX_VYY_NombreVideo" :var idioma="es" :var calidad="1080p60" :var musica="music/3Blue1Brown-Zeta-X3.mp3" :var volumen="0.25" :var volumen_voz="0.04" :results output
export LC_ALL=C   # decimales con punto para bc y printf
[ "$idioma" = "en" ] && raiz=../videoMath2 || raiz=.
dir=$raiz/lib/media/videos/$fichero/$calidad
tmp=$raiz/tmp/$video; mkdir -p $tmp; rm -f $tmp/seg_*.mp4

# 1. Un segmento por escena, con audio y vídeo de la misma duración exacta
i=0; lista=""; offsets=""; T=0; T0=""; Tc=""
for esc in $escenas; do
  seg=$(printf "%s/seg_%02d.mp4" $tmp $i)
  dur=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=duration -of default=nk=1:nw=1 $dir/$esc.mp4)
  fps=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=r_frame_rate -of default=nk=1:nw=1 $dir/$esc.mp4)
  if ffprobe -loglevel error -select_streams a -show_entries stream=codec_type -of csv=p=0 $dir/$esc.mp4 | grep -q audio; then
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -af "aresample=48000,apad" -ac 2 -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  else
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -f lavfi -i anullsrc=r=48000:cl=stereo -map 0:v -map 1:a -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  fi
  d=$(ffprobe -loglevel error -show_entries format=duration -of default=nk=1:nw=1 $seg)
  offsets="$offsets $T"; [ "$esc" = "$cierre" ] && Tc=$T
  printf "%2d  %-55s %8.3f s  (empieza en %8.3f)\n" $i $esc $d $T
  T=$(bc -l <<< "$T+$d"); [ -z "$T0" ] && T0=$d; dl=$d
  lista="$lista $seg"; i=$((i+1))
done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
# La voz se mide entera y recibe una ganancia fija hasta -16 LUFS, con un limitador para los picos.
# (loudnorm en una pasada ajusta la ganancia sobre la marcha: la primera frase quedaba 5 dB más baja.)
ia=""; for k in $(seq 0 $((n-1))); do ia="$ia[$k:a]"; done
I=$(ffmpeg -hide_banner -nostats $inputs -filter_complex "${ia}concat=n=$n:v=0:a=1,ebur128=framelog=quiet" -f null - 2>&1 | awk '$1=="I:"{v=$2} END{print v}')
ganancia=$(bc -l <<< "-16 - ($I)"); printf "Voz: %s LUFS; ganancia %.2f dB\n" $I $ganancia
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]volume=${ganancia}dB,alimiter=limit=0.84:level=false[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];
  [$((n+2)):a]atrim=$dur_m2:$fin_m3,asetpts=PTS-STARTPTS,volume=$volumen,afade=t=in:st=0:d=1,adelay=delays=${ms_m3}:all=1[m3];
  [voz][m1][m2][m3]amix=inputs=4:duration=first:normalize=0[mix];
  [mix]afade=t=out:st=$tfade:d=6[a];
  [v0]fade=t=out:st=$tfade:d=6[v]" \
  -map "[v]" -map "[a]" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -c:a aac -b:a 192k $raiz/$video.mp4
echo "Vídeo: $raiz/$video.mp4"

import sys, os, re

def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")

import ast, textwrap

arbol = ast.parse(open(f"lib/{fichero}.py", encoding="utf-8").read())
idx   = 0 if idioma == "es" else 1

def texto(nodo):
    try:
        return ast.literal_eval(nodo)
    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
