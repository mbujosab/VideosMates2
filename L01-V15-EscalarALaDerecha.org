** Utilidades del vídeo

=nace= y =entera= son las de V13 y V14 (cada expresión nace de la anterior). =viaje= es el viaje de ida

#+name: Utilidades del vídeo
#+BEGIN_SRC python :eval no
COLOR_SELECTOR = GREEN     # el selector, en verde; el escalar, en amarillo (COLOR_ESCALAR)

# Las reglas del vídeo: (lado izquierdo, lado derecho)
DER_VEC  = (r"\elemRp{\lambda\Vect{b}}{i}", r"\lambda\eleVRpE{b}{i}")      # ( λ b ) | i  =  λ ( b | i )
IZQ_VEC  = (r"\elemLp{\lambda\Vect{b}}{i}", r"\lambda\eleVLpE{b}{i}")      # i | ( λ b )  =  λ ( i | b )
IZQ_VEC2 = (r"\elemLp{\Vect{b}\lambda}{i}", r"\eleVLpE{b}{i}\lambda")      # i | ( b λ )  =  ( i | b ) λ
FIL_MAT  = (r"\elemLp{\Mat{A}\lambda}{i}", r"\VectFpE{A}{i}\lambda")       # i | ( A λ )  =  ( i | A ) λ
COL_MAT  = (r"\elemRp{\lambda\Mat{A}}{j}", r"\lambda\VectCpE{A}{j}")       # ( λ A ) | j  =  λ ( A | j )
TRA_MAT  = (r"\Transp{\Mat{A}\lambda}", r"\MatTpE{A}\lambda")              # ( A λ ) T  =  ( A T ) λ


def regla(par, font_size=48):
    return MathTex(par[0], "=", par[1], tex_template=myTemplate, font_size=font_size)


def fila_vec(fila):
    """Un vector numérico escrito en horizontal, entre paréntesis y con la coma final del libro."""
    return "(" + r",\;".join(str(x) for x in fila) + ",)"


def resalta(escena, mobs, color=COLOR_ESCALAR, factor=1.3, run_time=0.3):
    """Resaltado inmediato y sostenido (color y tamaño), sin la subida lenta de Indicate."""
    escena.play(*[x.animate.set_color(color).scale(factor) for x in mobs], run_time=run_time)


def apaga(escena, mobs, color=WHITE, factor=1.3, run_time=0.3):
    escena.play(*[x.animate.set_color(color).scale(1/factor) for x in mobs], run_time=run_time)


def viaje(escena, formula, mapa, run_time=2.4):
    """Los glifos del lado izquierdo de la regla viajan a la colocación del derecho y vuelven (V10, V13):
    mapa[i] es el glifo del lado derecho al que va el glifo i del izquierdo."""
    destino = formula[2].copy().move_to(formula[0])
    escena.play(*[Transform(formula[0][i], destino[j], rate_func=rate_functions.there_and_back_with_pause)
                  for i, j in enumerate(mapa)], run_time=run_time)


def rotulo(texto, junto_a, x, font_size=34):
    """Rótulo gris pequeño a la derecha de una línea (lo que justifica el paso)."""
    r = Tex(texto, tex_template=myTemplate, font_size=font_size, color=GRAY_A).next_to(junto_a, RIGHT, buff=0.6)
    return r.shift((x - r.get_left()[0])*RIGHT)


def nace(escena, origen, destino, fijos, moviles=(), aparecen=(), con_igual=True, extra=(), marca=None,
         t_fijos=0.8, t_moviles=0.9):
    """Una expresión nace de la anterior (CLAUDE.md, «Cada expresión nace de la anterior»). Primero, mientras la
    voz nombra la regla, se copia lo que no cambia (fijos: parejas glifo de origen, glifo de destino) y aparece el
    igual; después (en la marca, si se da) viaja lo que el paso mueve (moviles) y surge lo nuevo (aparecen)."""
    cubiertos = sorted({j for _, j in fijos} | {j for _, j in moviles} | set(aparecen) | ({0} if con_igual else set()))
    assert cubiertos == list(range(len(destino))), (cubiertos, len(destino))
    escena.play(*extra, *([FadeIn(destino[0])] if con_igual else []),
                *[TransformFromCopy(origen[i], destino[j]) for i, j in fijos], run_time=t_fijos)
    if marca is not None:
        escena.wait_until_bookmark(marca)
    if moviles or aparecen:
        escena.play(*[TransformFromCopy(origen[i], destino[j]) for i, j in moviles],
                    *[FadeIn(destino[j]) for j in aparecen], run_time=t_moviles)


def entera(escena, linea):
    """Tras nacer glifo a glifo, la línea vuelve a estar en la escena como un solo objeto."""
    escena.remove(*linea[0]); escena.add(linea)

*** Escena 1 - Arranque; el problema

Con el selector a la derecha, sacar el escalar es «desplazar los paréntesis»: los símbolos no cambian
de orden (se ve viajar la pareja de paréntesis). Con el selector a la izquierda, el escalar y el
selector intercambian sus posiciones: la imagen de los paréntesis ya no sirve.

*** Escena 3 - Demostración

La cadena de la Sección 1.4.4, $\elemLp{\Vect{b}\lambda}{i}=\elemLp{\lambda\Vect{b}}{i}=
\lambda\eleVLpE{b}{i}=\eleVLpE{b}{i}\lambda$, con un rótulo por paso (convenio; el selector deja fuera
al escalar; la conmutativa del producto de números). Después los dos extremos se recogen en la regla y los paréntesis
viajan: con el escalar a la derecha vuelve a bastar desplazarlos.

** Montaje: concatenar las escenas, música de entrada y salida, subtítulos

El vídeo final se monta a partir de los mp4 de las escenas (no de la
clase del vídeo completo): corregir una escena no obliga a renderizarlo
todo ni a pagar otra vez la voz, y los límites entre escenas quedan
exactos. Cada segmento se recodifica igualando la duración del audio y
del vídeo (concatenar sin recodificar deja un parón en cada unión, pues
manim genera el audio unas décimas más largo que la imagen). La música
suena en la portada (volumen =volumen=) y se funde cuando empieza la
primera escena; vuelve al empezar la escena =cierre= (bajo la voz, con
=volumen_voz=), sube en la última escena (los créditos) y se funde a
negro con la imagen. La voz se mide entera y se lleva a -16 LUFS con
una ganancia fija y un limitador (no con =loudnorm= en una pasada, que
subía el volumen poco a poco al empezar). Los =.srt= de las escenas se desplazan y se
funden en un único =video.srt= junto al mp4 (para subirlo a YouTube).

=escenas= es la lista ordenada de clases; =cierre= la escena en la que
vuelve la música (el resumen). El mp4 y el srt finales quedan en la raíz del repositorio del
idioma.

#+name: montaje
#+BEGIN_SRC bash :var fichero="LXX_VYY_FileName" :var escenas="L01_V15_E00_Portada L01_V15_E01_Problema L01_V15_E02_Convenio L01_V15_E03_Demostracion L01_V15_E04_Matrices L01_V15_E05_Resumen ZCreditosFinales" :var cierre="LXX_VYY_E05_Resumen" :var video="LXX_VYY_NombreVideo" :var idioma="es" :var calidad="1080p60" :var musica="music/3Blue1Brown-Zeta-X3.mp3" :var volumen="0.25" :var volumen_voz="0.04" :results output
export LC_ALL=C   # decimales con punto para bc y printf
[ "$idioma" = "en" ] && raiz=../videoMath2 || raiz=.
dir=$raiz/lib/media/videos/$fichero/$calidad
tmp=$raiz/tmp/$video; mkdir -p $tmp; rm -f $tmp/seg_*.mp4

# 1. Un segmento por escena, con audio y vídeo de la misma duración exacta
i=0; lista=""; offsets=""; T=0; T0=""; Tc=""
for esc in $escenas; do
  seg=$(printf "%s/seg_%02d.mp4" $tmp $i)
  dur=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=duration -of default=nk=1:nw=1 $dir/$esc.mp4)
  fps=$(ffprobe -loglevel error -select_streams v:0 -show_entries stream=r_frame_rate -of default=nk=1:nw=1 $dir/$esc.mp4)
  if ffprobe -loglevel error -select_streams a -show_entries stream=codec_type -of csv=p=0 $dir/$esc.mp4 | grep -q audio; then
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -af "aresample=48000,apad" -ac 2 -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  else
    ffmpeg -loglevel error -y -i $dir/$esc.mp4 -f lavfi -i anullsrc=r=48000:cl=stereo -map 0:v -map 1:a -t $dur -r $fps -fps_mode cfr -pix_fmt yuv420p -c:v libx264 -crf 18 -preset fast -c:a aac -b:a 192k $seg
  fi
  d=$(ffprobe -loglevel error -show_entries format=duration -of default=nk=1:nw=1 $seg)
  offsets="$offsets $T"; [ "$esc" = "$cierre" ] && Tc=$T
  printf "%2d  %-55s %8.3f s  (empieza en %8.3f)\n" $i $esc $d $T
  T=$(bc -l <<< "$T+$d"); [ -z "$T0" ] && T0=$d; dl=$d
  lista="$lista $seg"; i=$((i+1))
done
n=$i; Tl=$(bc -l <<< "$T-$dl"); [ -z "$Tc" ] && Tc=$Tl
printf "Duración total %.3f s; música hasta %.3f s, desde %.3f s (bajo la voz) y desde %.3f s (créditos)\n" $T $T0 $Tc $Tl

# 2. Concatenación recodificada, voz normalizada, música de entrada y salida, fundido final
inputs=""; fc=""; k=0
for seg in $lista; do inputs="$inputs -i $seg"; fc="$fc[$k:v][$k:a]"; k=$((k+1)); done
# La voz se mide entera y recibe una ganancia fija hasta -16 LUFS, con un limitador para los picos.
# (loudnorm en una pasada ajusta la ganancia sobre la marcha: la primera frase quedaba 5 dB más baja.)
ia=""; for k in $(seq 0 $((n-1))); do ia="$ia[$k:a]"; done
I=$(ffmpeg -hide_banner -nostats $inputs -filter_complex "${ia}concat=n=$n:v=0:a=1,ebur128=framelog=quiet" -f null - 2>&1 | awk '$1=="I:"{v=$2} END{print v}')
ganancia=$(bc -l <<< "-16 - ($I)"); printf "Voz: %s LUFS; ganancia %.2f dB\n" $I $ganancia
fin_m1=$(bc -l <<< "$T0+5"); tfade=$(bc -l <<< "$T-6")
dur_m2=$(bc -l <<< "$Tl-$Tc"); ms_m2=$(printf "%.0f" $(bc -l <<< "$Tc*1000")); fade_m2=$(bc -l <<< "$dur_m2-1")
fin_m3=$(bc -l <<< "$T-$Tc"); ms_m3=$(printf "%.0f" $(bc -l <<< "$Tl*1000"))
ffmpeg -loglevel error -y $inputs -i $musica -i $musica -i $musica -filter_complex "
  ${fc}concat=n=$n:v=1:a=1[v0][a0];
  [a0]volume=${ganancia}dB,alimiter=limit=0.84:level=false[voz];
  [$n:a]atrim=0:$fin_m1,volume=$volumen,afade=t=in:st=0:d=1,afade=t=out:st=$T0:d=5[m1];
  [$((n+1)):a]atrim=0:$dur_m2,volume=$volumen_voz,afade=t=in:st=0:d=3,afade=t=out:st=$fade_m2:d=1,adelay=delays=${ms_m2}:all=1[m2];
  [$((n+2)):a]atrim=$dur_m2:$fin_m3,asetpts=PTS-STARTPTS,volume=$volumen,afade=t=in:st=0:d=1,adelay=delays=${ms_m3}:all=1[m3];
  [voz][m1][m2][m3]amix=inputs=4:duration=first:normalize=0[mix];
  [mix]afade=t=out:st=$tfade:d=6[a];
  [v0]fade=t=out:st=$tfade:d=6[v]" \
  -map "[v]" -map "[a]" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -c:a aac -b:a 192k $raiz/$video.mp4
echo "Vídeo: $raiz/$video.mp4"

# 3. Subtítulos: desplazar el srt de cada escena y fundirlos en uno
ESCENAS="$escenas" OFFSETS="$offsets" python3 - "$dir" "$raiz/$video.srt" <<'EOF'
import sys, os, re
dir, salida = sys.argv[1], sys.argv[2]
escenas = os.environ["ESCENAS"].split(); offsets = [float(x) for x in os.environ["OFFSETS"].split()]
def t2s(t):
    h, m, s = t.strip().split(":"); s, ms = s.split(",")
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000
def s2t(x):
    ms = round(x*1000); h, r = divmod(ms, 3600000); m, r = divmod(r, 60000); s, ms = divmod(r, 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"
k, out = 1, []
for esc, off in zip(escenas, offsets):
    ruta = f"{dir}/{esc}.srt"
    if not os.path.exists(ruta): continue
    for bloque in re.split(r"\n\s*\n", open(ruta, encoding="utf-8").read().strip()):
        lineas = bloque.strip().splitlines()
        if len(lineas) < 3 or "-->" not in lineas[1]: continue
        a, _, b = lineas[1].partition("-->")
        out.append(f"{k}\n{s2t(t2s(a)+off)} --> {s2t(t2s(b)+off)}\n" + "\n".join(lineas[2:])); k += 1
open(salida, "w", encoding="utf-8").write("\n\n".join(out) + "\n")
print(f"Subtítulos: {salida} ({k-1} entradas)")
EOF

    except Exception:
        return ast.unparse(nodo)

def es_tr(n):
    return (isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
            and n.func.id == "tr" and len(n.args) >= 2)

def es_borrado(stmt):
    return ast.unparse(stmt).startswith("self.play(*[FadeOut(mob) for mob in self.mobjects])")

# Primera clase con construct(): la del vídeo completo.
construct = next(f for c in arbol.body if isinstance(c, ast.ClassDef)
                 for f in c.body if isinstance(f, ast.FunctionDef) and f.name == "construct")

escena, items = 0, []   # la portada es la escena 0
def volcar():
    global items
    if items:
        print(f"==== {'Escena' if idioma == 'es' else 'Scene'} {escena} ====\n")
        for tipo, t in items:
            if tipo == "voz":
                print(textwrap.fill(t, 72), "\n")
            else:
                print(f"    [{t}]\n")
    items = []

for stmt in construct.body:
    if es_borrado(stmt):
        volcar(); escena += 1; continue
    en_voz = set()
    for n in ast.walk(stmt):
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == "voiceover":
            en_voz |= {id(s) for s in ast.walk(n) if es_tr(s)}
    for n in sorted((n for n in ast.walk(stmt) if es_tr(n)), key=lambda n: (n.lineno, n.col_offset)):
        items.append(("voz" if id(n) in en_voz else "pantalla", texto(n.args[idx])))
volcar()
